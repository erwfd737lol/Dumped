local function myChar()
    local live = Workspace:FindFirstChild("Live")
    if live then
        local found = live:FindFirstChild(LP.Name)
        if found then
            return found
        end
    end
    return LP.Character
end

local function myRoot()
    local char = myChar()
    if not char then
        return nil
    end
    return char:FindFirstChild("HumanoidRootPart")
end

local function myHum()
    local char = myChar()
    if not char then
        return nil
    end
    return char:FindFirstChildOfClass("Humanoid")
end

local function rootOf(plr)
    local char = plr.Character
    if not char then
        return nil
    end
    return char:FindFirstChild("HumanoidRootPart")
end

local function humOf(plr)
    local char = plr.Character
    if not char then
        return nil
    end
    return char:FindFirstChildOfClass("Humanoid")
end

local function alive(plr)
    local hum = humOf(plr)
    if not hum then
        return false
    end
    if hum.Health <= 0 then
        return false
    end
    return rootOf(plr) ~= nil
end

local function allRooms()
    local out = {}
    for _, folderName in ipairs(ROOM_FOLDERS) do
        local folder = Workspace:FindFirstChild(folderName)
        if folder then
            for _, room in ipairs(folder:GetChildren()) do
                table.insert(out, room)
            end
        end
    end
    return out
end

local function factionOf(plr)
    local root = rootOf(plr)
    if not root then
        return nil
    end
    local gui = root:FindFirstChild("Faction")
    if not gui then
        return nil
    end
    local label = gui:FindFirstChild("FactionName")
    if not label then
        return nil
    end
    return label.Text
end

local function myFaction()
    return factionOf(LP)
end

local function armorOf(plr)
    local root = rootOf(plr)
    if not root then
        return nil
    end
    local value = root:GetAttribute("Armor")
    if type(value) ~= "number" then
        return nil
    end
    return value
end

local PART_ORDER = {
    Head  = { "Head", "UpperTorso", "HumanoidRootPart", "LowerTorso", "LeftUpperArm", "RightUpperArm" },
    Torso = { "UpperTorso", "LowerTorso", "Torso", "HumanoidRootPart", "Head" },

local function partsOf(plr, mode)
    local out = {}
    local char = plr.Character
    if not char then
        return out
    end
    local order = PART_ORDER[mode]
    if not order then
        order = PART_ORDER.Head
    end
    for _, partName in ipairs(order) do
        local part = char:FindFirstChild(partName)
        if part then
            if part:IsA("BasePart") then
                table.insert(out, part)
            end
        end
    end
    return out
end

local function fbApply()
    Lighting.Ambient = Color3.new(1, 1, 1)
    Lighting.ColorShift_Bottom = Color3.new(1, 1, 1)
    Lighting.ColorShift_Top = Color3.new(1, 1, 1)
    Lighting.Brightness = 3
    Lighting.ClockTime = 14
    Lighting.FogEnd = 1e6
    Lighting.GlobalShadows = false

    local atmosphere = Lighting:FindFirstChildOfClass("Atmosphere")
    if atmosphere then
        if atmosphere.Parent then
            if not S.fb.saved.atm then
                S.fb.saved.atm = atmosphere
            end
            atmosphere.Parent = nil
        end
    end
end

local function setFullBright(on)
    Flags.FullBright = on
    syncUI("FullBright", on)

    if on then
        if S.fb.active then
            return
        end
        S.fb.active = true
        S.fb.saved = {
            ambient = Lighting.Ambient,
            shiftB = Lighting.ColorShift_Bottom,
            shiftT = Lighting.ColorShift_Top,
            brightness = Lighting.Brightness,
            clock = Lighting.ClockTime,
            fogEnd = Lighting.FogEnd,
            shadows = Lighting.GlobalShadows,
            atm = Lighting:FindFirstChildOfClass("Atmosphere"),
        }
        fbApply()
        S.fb.conn = Lighting.LightingChanged:Connect(function()
            if S.fb.active then
                fbApply()
            end
        end)
        notify("FullBright", "Enabled")
        return
    end

    if not S.fb.active then
        return
    end
    S.fb.active = false
    safeDisconnect(S.fb.conn)
    S.fb.conn = nil

    local saved = S.fb.saved
    if saved then
        Lighting.Ambient = saved.ambient
        Lighting.ColorShift_Bottom = saved.shiftB
        Lighting.ColorShift_Top = saved.shiftT
        Lighting.Brightness = saved.brightness
        Lighting.ClockTime = saved.clock
        Lighting.FogEnd = saved.fogEnd
        Lighting.GlobalShadows = saved.shadows
        if saved.atm then
            saved.atm.Parent = Lighting
        end
    end
    S.fb.saved = nil
    notify("FullBright", "Disabled")
end

onDestroy(function()
    safeDisconnect(S.fb.conn)
end)

local function zeroPrompt(prompt)
    if not prompt then
        return
    end
    prompt.HoldDuration = 0
    prompt.MaxActivationDistance = math.max(prompt.MaxActivationDistance, 30)
end

local function sweepPrompts(doors, pickup)
    for _, room in ipairs(allRooms()) do
        if doors then
            local door = room:FindFirstChild("Door")
            local handle = door and door:FindFirstChild("Handle")
            if handle then
                zeroPrompt(handle:FindFirstChild("ProximityPrompt"))
            end
        end
        if pickup then
            local contents = room:FindFirstChild("Contents")
            local loot = contents and contents:FindFirstChild("Loot")
            if loot then
                for _, item in ipairs(loot:GetChildren()) do
                    zeroPrompt(item:FindFirstChildOfClass("ProximityPrompt"))
                end
            end
        end
    end
end

for _, folderName in ipairs(ROOM_FOLDERS) do
    local folder = Workspace:FindFirstChild(folderName)
    if folder then
        track(folder.DescendantAdded:Connect(function(descendant)
            if not descendant:IsA("ProximityPrompt") then
                return
            end
            local parent = descendant.Parent
            local isDoor = parent and parent.Name == "Handle"
            if isDoor and Flags.instantOpen then
                zeroPrompt(descendant)
                return
            end
            if not isDoor and Flags.instantPickup then
                zeroPrompt(descendant)
            end
        end))
    end
end

local function setRoll(on)
    Flags.rollSell = on
    syncUI("infMoney", on)
    if not on then
        return
    end
    notify("Roll & Sell", "Started")

    task.spawn(function()
        while Flags.rollSell do
            local fn = remoteFunc()
            local event = remoteEvent()
            if fn then
                pcall(fn.InvokeServer, fn, "PremRollSkin10")
            end
            if event then
                for _, weapon in ipairs(SELL_WEAPONS) do
                    if not Flags.rollSell then
                        break
                    end
                    for _, skin in ipairs(SELL_SKINS) do
                        pcall(event.FireServer, event, "SellSkin", weapon, skin)
                        pcall(event.FireServer, event, "SellSkin", weapon, skin)
                    end
                    task.wait()
                end
            end
            task.wait(math.max(Flags.rollInterval, 0.3))
        end
        notify("Roll & Sell", "Stopped")
    end)
end

local function nativeApply(inst, kind, on)
    local tags = NATIVE_TAGS[kind]
    if not tags then
        return false
    end
    local found = false
    for _, tag in ipairs(tags) do
        for _, obj in ipairs(inst:GetDescendants()) do
            if obj.Name == tag then
                local stored = S.nativeApplied[obj]
                if on then
                    if not stored then
                        stored = { enabled = obj.Enabled }
                        S.nativeApplied[obj] = stored
                    end
                    obj.Enabled = true
                    found = true
                else
                    if stored then
                        obj.Enabled = stored.enabled
                        S.nativeApplied[obj] = nil
                    end
                end
            end
        end
    end
    return found
end

local function clearEsp(inst)
    local record = S.espObjs[inst]
    if not record then
        return
    end
    if record.hum then
        pcall(function()
            record.hum.DisplayDistanceType = record.distPrev
        end)
    end
    if record.hl then
        record.hl:Destroy()
    end
    if record.tracer then
        pcall(record.tracer.Remove, record.tracer)
    end
    if record.gui then
        record.gui:Destroy()
    end
    S.espObjs[inst] = nil
end

local function clearAllEsp()
    for inst in pairs(S.espObjs) do
        clearEsp(inst)
    end
    for obj in pairs(S.nativeApplied) do
        pcall(function()
            obj.Enabled = S.nativeApplied[obj].enabled
        end)
        S.nativeApplied[obj] = nil
    end
end

local function espPos(inst)
    if inst:IsA("BasePart") then
        return inst.Position
    end
    if inst:IsA("Model") then
        local part = inst.PrimaryPart
        if not part then
            part = inst:FindFirstChildWhichIsA("BasePart")
        end
        if part then
            return part.Position
        end
    end
    return nil
end

local function ownRooms()
    local out = {}
    local me = LP.Name
    for _, room in ipairs(allRooms()) do
        local owner = room:GetAttribute("Owner")
        if owner == me then
            table.insert(out, room)
        elseif owner == LP.UserId then
            table.insert(out, room)
        end
    end
    return out
end

local function wantedEsp()
    local want = {}
    local mine = myFaction()

    if Flags.playersESP then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and alive(plr) then
                local same = false
                if mine then
                    same = factionOf(plr) == mine
                end
                if not (Flags.teamCheck and same) then
                    want[plr.Character] = "Player"
                end
            end
        end
    end

    if Flags.enemiesESP then
        local mobs = Workspace:FindFirstChild("Mobs")
        if mobs then
            for _, mob in ipairs(mobs:GetChildren()) do
                want[mob] = "Mob"
            end
        end
    end

    if Flags.itemsESP then
        for _, room in ipairs(allRooms()) do
            local contents = room:FindFirstChild("Contents")
            local loot = contents and contents:FindFirstChild("Loot")
            if loot then
                for _, item in ipairs(loot:GetChildren()) do
                    if rarityRank(item.Name) >= Flags.itemMinRank then
                        want[item] = "Item"
                    end
                end
            end
        end
    end

    if Flags.baseESP then
        for _, room in ipairs(ownRooms()) do
            want[room] = "Base"
        end
    end

    if Flags.elevatorESP then
        local elevator = Workspace:FindFirstChild("Elevator")
        if elevator then
            for _, child in ipairs(elevator:GetChildren()) do
                want[child] = "Elevator"
            end
        end
    end

    return want
end

local function reconcileEsp()
    local want = wantedEsp()

    for inst in pairs(S.espObjs) do
        if not want[inst] then
            clearEsp(inst)
        elseif not inst.Parent then
            clearEsp(inst)
        end
    end

    if Flags.nativeEsp then
        for inst, kind in pairs(want) do
            local applied = nativeApply(inst, kind, true)
            if not applied then
                -- no game object for this kind: fall back to the overlay
                if not S.espObjs[inst] then
                    buildEsp(inst, kind)
                end
            elseif S.espObjs[inst] then
                clearEsp(inst)
            end
        end
        return
    end

    for inst, kind in pairs(want) do
        if not S.espObjs[inst] then
            buildEsp(inst, kind)
        end
    end
end

local function tintFor(inst, kind)
    if kind == "Player" then
        local plr = Players:GetPlayerFromCharacter(inst)
        local faction = plr and factionOf(plr)
        if faction then
            local color = FACTION_COLORS[faction]
            if color then
                return color
            end
        end
        return Color3.fromRGB(235, 235, 235)
    end
    if kind == "Mob" then
        return Color3.fromRGB(255, 130, 110)
    end
    if kind == "Item" then
        return rarityColor(inst.Name)
    end
    if kind == "Base" then
        return Color3.fromRGB(120, 235, 150)
    end
    return Color3.fromRGB(180, 200, 255)
end

local function updateEsp()
    local camera = cam()
    if not camera then
        return
    end
    local camPos = camera.CFrame.Position
    local viewport = camera.ViewportSize

    for inst, record in pairs(S.espObjs) do
        local pos = espPos(inst)
        if pos then
            local dist = (pos - camPos).Magnitude
            local near = dist <= Flags.espDist
            local tint = tintFor(inst, record.kind)

            record.gui.Enabled = near
            record.stroke.Color = tint
            record.name.Text = inst.Name
            record.name.TextColor3 = tint
            record.name.Visible = false
            if Flags.nameTags then
                if record.kind ~= "Item" then
                    record.name.Visible = true
                end
            end

            record.sub.Visible = record.kind == "Player"
            if record.kind == "Player" then
                local plr = Players:GetPlayerFromCharacter(inst)
                local label = ""
                if plr then
                    label = factionOf(plr) or ""
                    local armor = armorOf(plr)
                    if armor and armor > 0 then
                        label = fmt("%s · %dA", label, armor)
                    end
                end
                record.sub.Text = label
                record.sub.TextColor3 = tint
            end

            record.dist.Text = fmt("%dm", math.floor(dist))

            if record.bar then
                local hum = inst:FindFirstChildOfClass("Humanoid")
                record.barBg.Visible = false
                if hum and Flags.healthBars then
                    record.barBg.Visible = true
                    local ratio = hum.Health / math.max(hum.MaxHealth, 1)
                    ratio = math.clamp(ratio, 0, 1)
                    record.bar.Size = UDim2.fromScale(ratio, 1)
                    record.bar.BackgroundColor3 = Color3.fromRGB(
                        math.floor(255 * (1 - ratio) + 60),
                        math.floor(60 + 180 * ratio),
                        90)
                end
            end

            if record.hl then
                record.hl.Enabled = near
                record.hl.FillColor = tint
            end

            if record.tracer then
                record.tracer.Visible = false
                if Flags.tracers and near then
                    local screen, onScreen = camera:WorldToViewportPoint(pos)
                    record.tracer.Visible = onScreen
                    record.tracer.From = Vector2.new(viewport.X * 0.5, viewport.Y)
                    record.tracer.To = Vector2.new(screen.X, screen.Y)
                    record.tracer.Color = tint
                end
            end
        else
            record.gui.Enabled = false
        end
    end
end

local function manipPoint(base, origin)
    local point = base
    if not Flags.manip then
        return point
    end
    if Flags.manipV ~= 0 then
        point = point + Vector3.yAxis * Flags.manipV
    end
    if Flags.manipL ~= 0 then
        local camera = cam()
        point = point + camera.CFrame.RightVector * Flags.manipL
    end
    if Flags.manipD ~= 0 then
        local delta = point - origin
        if delta.Magnitude > 0.01 then
            point = point + delta.Unit * Flags.manipD
        end
    end
    return point
end

local function castSegment(from, to, targetChar)
    local ignore = {}
    local mine = myChar()
    if mine then
        table.insert(ignore, mine)
    end
    if targetChar then
        table.insert(ignore, targetChar)
    end
    S.params.FilterDescendantsInstances = ignore
    return Workspace:Raycast(from, to - from, S.params)
end

local function solve(plr)
    local char = plr.Character
    if not char then
        return nil, false
    end

    local origin = muzzle()
    local best = nil
    local bestScore = -math.huge
    local clear = false

    for _, part in ipairs(partsOf(plr, Flags.combatPart)) do
        local point = manipPoint(part.Position, origin)
        local hit = castSegment(origin, point, char)
        local score = math.huge
        if hit then
            score = hit.Distance
        end
        if score > bestScore then
            bestScore = score
            best = point
        end
        if not hit then
            clear = true
            if score == math.huge then
                break
            end
        end
    end

    return best, clear
end

local function passesFilter(plr)
    if plr == LP then
        return false
    end
    if not alive(plr) then
        return false
    end
    local mine = myFaction()
    if mine then
        if factionOf(plr) == mine then
            return false
        end
    end
    return true
end

local function scan()
    local camera = cam()
    if not camera then
        return nil
    end

    local camPos = camera.CFrame.Position
    local look = camera.CFrame.LookVector
    local best = nil
    local bestScore = math.huge

    for _, plr in ipairs(Players:GetPlayers()) do
        if passesFilter(plr) then
            local part = partsOf(plr, Flags.combatPart)[1]
            if part then
                local vel = part.AssemblyLinearVelocity
                local pos = part.Position + vel * (Flags.prediction / 100)
                local delta = pos - camPos
                local dist = delta.Magnitude
                if dist > 0.5 and dist <= Flags.maxCombatDist then
                    local dot = delta.Unit:Dot(look)
                    local angle = math.deg(math.acos(math.clamp(dot, -1, 1)))
                    if angle <= Flags.combatFov then
                        if angle < bestScore then
                            bestScore = angle
                            best = { plr = plr, pos = pos }
                        end
                    end
                end
            end
        end
    end

    return best
end

local function refreshSilent()
    if not Flags.silent then
        S.silentPoint = nil
        S.silentClear = false
        S.silentTarget = nil
        return
    end

    local target = scan()
    if not target then
        S.silentPoint = nil
        S.silentClear = false
        S.silentTarget = nil
        return
    end

    local point, clear = solve(target.plr)
    S.silentTarget = target
    S.silentClear = clear
    if point then
        S.silentPoint = point
    else
        S.silentPoint = target.pos
    end
end

local ATTACK_ACTIONS = { M1D = true, M2D = true }

local function rewriteAimArgs(args)
    local point = S.silentPoint
    if not point then
        return nil
    end
    for index = 2, args.n do
        local valueType = typeof(args[index])
        if valueType == "CFrame" then
            args[index] = CFrame.new(muzzle(), point)
            return args
        end
        if valueType == "Vector3" then
            local delta = point - muzzle()
            if delta.Magnitude > 0.05 then
                args[index] = delta.Unit
                return args
            end
        end
    end
    return nil
end

local function installHooks()
    if not hookmetamethod then
        return
    end

    local ok, old = pcall(function()
        return hookmetamethod(game, "__namecall", function(self, ...)
            local method = getnamecallmethod()
            if method ~= "FireServer" then
                return S.oldNamecall(self, ...)
            end
            if typeof(self) ~= "Instance" then
                return S.oldNamecall(self, ...)
            end

            if self.Name == "UnreliableRemoteEvent" then
                if Flags.spoofAim then
                    local args = table.pack(...)
                    if args[1] == "ReportAim" then
                        local rewritten = rewriteAimArgs(args)
                        if rewritten then
                            return S.oldNamecall(self, table.unpack(rewritten, 1, rewritten.n))
                        end
                    end
                end
                return S.oldNamecall(self, ...)
            end

            if self.Name ~= "RemoteEvent" then
                return S.oldNamecall(self, ...)
            end
            if not Flags.silent then
                return S.oldNamecall(self, ...)
            end

            local args = table.pack(...)
            local action = args[1]
            if typeof(action) ~= "string" then
                return S.oldNamecall(self, ...)
            end
            if not ATTACK_ACTIONS[action] then
                return S.oldNamecall(self, ...)
            end

            local rewritten = rewriteAimArgs(args)
            if rewritten then
                return S.oldNamecall(self, table.unpack(rewritten, 1, rewritten.n))
            end
            return S.oldNamecall(self, ...)
        end)
    end)
    if ok then
        S.oldNamecall = old
    end

    local ok2, old2 = pcall(function()
        return hookmetamethod(game, "__newindex", function(self, key, value)
            if key ~= "WalkSpeed" then
                return S.oldNewIndex(self, key, value)
            end
            if not Flags.Speed then
                return S.oldNewIndex(self, key, value)
            end
            if typeof(self) ~= "Instance" then
                return S.oldNewIndex(self, key, value)
            end
            if not self:IsA("Humanoid") then
                return S.oldNewIndex(self, key, value)
            end
            if self.Parent ~= myChar() then
                return S.oldNewIndex(self, key, value)
            end

            local want
            if Flags.speedMode == "Set" then
                want = Flags.speedValue
            elseif Flags.speedMode == "Multiply" then
                want = 12 * Flags.speedValue
            else
                want = 12 + Flags.speedValue
            end
            if math.abs(value - want) > 0.01 then
                value = want
            end

            return S.oldNewIndex(self, key, value)
        end)
    end)
    if ok2 then
        S.oldNewIndex = old2
        S.hookOK = true
    end
end

installHooks()

local function fireAttack(action)
    local event = remoteEvent()
    if not event then
        return false
    end
    if Flags.requireClear then
        if S.silentTarget and not S.silentClear then
            return false
        end
    end

    local origin = muzzle()
    local lookAt = S.silentPoint
    if not lookAt then
        local camera = cam()
        lookAt = camera.CFrame.Position + camera.CFrame.LookVector * 200
    end
    if (lookAt - origin).Magnitude < 0.05 then
        lookAt = origin + Vector3.yAxis * 0.5
    end

    pcall(event.FireServer, event, action, CFrame.new(origin, lookAt))

    local releaseDelay = 0.06
    if action == "M2D" then
        releaseDelay = 0.08
    end
    task.delay(releaseDelay, function()
        local again = remoteEvent()
        if again then
            pcall(again.FireServer, again, action:gsub("D", "U"))
        end
    end)

    return true
end

local function attackLoop()
    if S.attacking then
        return
    end
    S.attacking = true
    while Flags.autoAttack do
        local char = myChar()
        local tool = char and char:FindFirstChildOfClass("Tool")
        local weapon = tool and WEAPON_BY_NAME[tool.Name]
        local cooldown = 0.12
        if weapon then
            cooldown = weapon.cd
        end
        fireAttack("M1D")
        task.wait(math.max(cooldown, 0.05))
    end
    S.attacking = false
end

local function setAttack(on)
    Flags.autoAttack = on
    syncUI("autoAttack", on)
    if on then
        task.spawn(attackLoop)
    end
end

local flyBV = nil
local flyBG = nil

local function setFly(on)
    Flags.Fly = on
    syncUI("Fly", on)

    if on then
        local root = myRoot()
        if not root then
            Flags.Fly = false
            syncUI("Fly", false)
            return
        end
        local hum = myHum()
        if hum then
            hum.PlatformStand = true
        end

        flyBV = Instance.new("BodyVelocity")
        flyBV.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        flyBV.Velocity = Vector3.zero
        flyBV.Parent = root

        flyBG = Instance.new("BodyGyro")
        flyBG.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
        flyBG.P = 2500
        flyBG.Parent = root
        return
    end

    if flyBV then
        flyBV:Destroy()
        flyBV = nil
    end
    if flyBG then
        flyBG:Destroy()
        flyBG = nil
    end
    local hum = myHum()
    if hum then
        hum.PlatformStand = false
    end
end

local function flyStep()
    if not Flags.Fly then
        return
    end
    if not flyBV then
        return
    end
    if not flyBV.Parent then
        return
    end

    local camera = cam()
    local dir = Vector3.zero

    if UIS:IsKeyDown(Enum.KeyCode.W) then
        dir = dir + camera.CFrame.LookVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.S) then
        dir = dir - camera.CFrame.LookVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.A) then
        dir = dir - camera.CFrame.RightVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.D) then
        dir = dir + camera.CFrame.RightVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.Space) then
        dir = dir + Vector3.yAxis
    end
    if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then
        dir = dir - Vector3.yAxis
    end

    if dir.Magnitude > 0 then
        flyBV.Velocity = dir.Unit * Flags.flySpeed
    else
        flyBV.Velocity = Vector3.zero
    end
    flyBG.CFrame = camera.CFrame
end

local function setCollisions(char, on)
    if not char then
        return
    end
    for _, descendant in ipairs(char:GetDescendants()) do
        if descendant:IsA("BasePart") then
            descendant.CanCollide = on
        end
    end
end

local function farmTargets()
    local out = {}
    local names = Flags.farmNames
    local takeAll = next(names) == nil

    for _, room in ipairs(allRooms()) do
        local folder = room.Parent
        if folder then
            if Flags.farmScope[folder.Name] then
                local contents = room:FindFirstChild("Contents")
                local loot = contents and contents:FindFirstChild("Loot")
                if loot then
                    for _, item in ipairs(loot:GetChildren()) do
                        local wanted = takeAll
                        if not wanted then
                            local lowered = item.Name:lower()
                            for name in pairs(names) do
                                if lowered:find(name:lower(), 1, true) then
                                    wanted = true
                                    break
                                end
                            end
                        end
                        if wanted then
                            local pos = espPos(item)
                            if pos then
                                table.insert(out, { inst = item, pos = pos })
                            end
                        end
                    end
                end
            end
        end
    end

    return out
end

local function stopFarm()
    S.farmRunning = false
    if warnFrame then
        warnFrame.Visible = false
    end
    pcall(CAS.UnbindAction, CAS, "AdvFarmLock")
    setCollisions(myChar(), true)
    if Flags.returnOnStop then
        if S.farmOrigin then
            local root = myRoot()
            if root then
                root.CFrame = S.farmOrigin
            end
        end
    end
end

local function farmLoop()
    while S.farmRunning do
        local targets = farmTargets()
        if #targets == 0 then
            task.wait(0.4)
        else
            for _, target in ipairs(targets) do
                if not S.farmRunning then
                    break
                end
                local root = myRoot()
                if not root then
                    break
                end
                if not target.inst.Parent then
                    break
                end

                root.CFrame = CFrame.new(target.pos + Vector3.yAxis * 2.4)

                local prompt = target.inst:FindFirstChildOfClass("ProximityPrompt")
                if prompt then
                    pcall(prompt.InputHoldBegin, prompt)
                    task.delay(0.08, function()
                        pcall(prompt.InputHoldEnd, prompt)
                    end)
                end

                task.wait(Flags.farmInterval)
            end
        end
    end
    stopFarm()
end

local function refreshFarm()
    if not Flags.farmEnabled then
        if S.farmRunning then
            stopFarm()
            notify("Farm", "Stopped")
        end
        return
    end

    if next(Flags.farmScope) == nil then
        if S.farmRunning then
            stopFarm()
        end
        return
    end

    if S.farmRunning then
        return
    end

    local root = myRoot()
    if not root then
        notify("Farm", "no character")
        return
    end
    S.farmRunning = true
    S.farmOrigin = root.CFrame
    setCollisions(myChar(), false)
    CAS:BindActionAtPriority("AdvFarmLock", function() end, false, 3000,
        Enum.PlayerActions.CharacterForward,
        Enum.PlayerActions.CharacterBackward,
        Enum.PlayerActions.CharacterLeft,
        Enum.PlayerActions.CharacterRight,
        Enum.PlayerActions.CharacterJump)
    showWarning()
    task.spawn(farmLoop)
    notify("Farm", "Started")
end

local function watch(key, name, on, isMob)
    safeDisconnect(S.watchers[key])
    S.watchers[key] = nil
    if not on then
        return
    end

    for _, folderName in ipairs(ROOM_FOLDERS) do
        local folder = Workspace:FindFirstChild(folderName)
        if folder then
            S.watchers[key] = folder.DescendantAdded:Connect(function(descendant)
                if descendant.Name ~= name then
                    return
                end
                if isMob then
                    local parent = descendant.Parent
                    if parent then
                        if parent.Name == "Mobs" then
                            notify("Mob", name)
                        end
                    end
                    return
                end
                notify("NPC", name)
            end)
            return
        end
    end
end

local function checkBounty(plr)
    if not Flags.bountyAlert then
        return
    end
    local stats = plr:FindFirstChild("leaderstats")
    local bounty = stats and stats:FindFirstChild("Bounty")
    if not bounty then
        return
    end
    if type(bounty.Value) ~= "number" then
        return
    end
    if bounty.Value >= 1000 then
        notify("Bounty", fmt("%s — %d", plr.Name, bounty.Value))
    end
end

local function goBase()
    for _, room in ipairs(ownRooms()) do
        local floor = room:FindFirstChild("Floor")
        if floor then
            teleport(floor.CFrame + Vector3.new(0, 3, 0))
            notify("Travel", "Base")
            return
        end
    end
    notify("Travel", "no owned storage")
end

local function goElevator()
    local elevator = Workspace:FindFirstChild("Elevator")
    local box = elevator and elevator:FindFirstChild("Box")
    local doors = box and box:FindFirstChild("Doors")
    if doors then
        teleport(doors:GetPivot())
    else
        teleport(TRAVEL_ELEVATOR)
    end
    notify("Travel", "Elevator")
end

local function goNpc(name, offset)
    local char = myChar()
    if not char then
        return
    end
    for _, room in ipairs(allRooms()) do
        local contents = room:FindFirstChild("Contents")
        local folder = contents and contents:FindFirstChild("NPC")
        local npc = folder and folder:FindFirstChild(name)
        if npc then
            local pivot = npc:GetPivot()
            local behind = pivot.LookVector * offset
            char:PivotTo(CFrame.new(pivot.Position + behind + Vector3.new(0, 1.5, 0)))
            notify("Travel", name)
            return
        end
    end
    notify("Travel", name .. " not found")
end

local function goPlayer(name)
    local plr = Players:FindFirstChild(name)
    if not plr then
        notify("Travel", "unavailable")
        return
    end
    if not alive(plr) then
        notify("Travel", "unavailable")
        return
    end
    local root = rootOf(plr)
    if root then
        teleport(root.CFrame + Vector3.new(0, 2, 0))
    end
end

local function onSaveWaypoint()

    local name = option and option.Value
    local root = myRoot()
    if not root then
        return

        notify("Waypoint", "name required")
        return
    end
    if name == "" then
        notify("Waypoint", "name required")
        return
    end
    S.waypoints[name] = root.CFrame
    refreshWaypoints()
    notify("Waypoint", name .. " saved")
end

local function heartStep(dt)
    if Flags.autoRun then
        local char = myChar()
        local event = remoteEvent()
        if char then
            char:SetAttribute("Running", true)
        end
        if event then
            pcall(event.FireServer, event, "Run")
        end
    end

    if Flags.Noclip then
        local char = myChar()
        if char then
            for _, descendant in ipairs(char:GetDescendants()) do
                if descendant:IsA("BasePart") then
                    if descendant.CanCollide then
                        descendant.CanCollide = false
                    end
                end
            end
        end
    end

    if Flags.Speed then
        local hum = myHum()
        if hum then
            local want
            if Flags.speedMode == "Set" then
                want = Flags.speedValue
            elseif Flags.speedMode == "Multiply" then
                want = 12 * Flags.speedValue
            else
                want = 12 + Flags.speedValue
            end
            if math.abs(hum.WalkSpeed - want) > 0.01 then
                hum.WalkSpeed = want
            end
        end
    end

    if Flags.autoReload then
        local char = myChar()
        local tool = char and char:FindFirstChildOfClass("Tool")
        local event = remoteEvent()
        if tool and event then
            if WEAPON_BY_NAME[tool.Name] then
                pcall(event.FireServer, event, "Reload")
            end
        end
    end

    if Flags.autoKnife then
        knifeIfEmpty()
    end

    if Flags.autoBlock then
        local root = myRoot()
        local threat = false
        if root then
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LP and alive(plr) then
                    local other = rootOf(plr)
                    if other then
                        if (other.Position - root.Position).Magnitude <= Flags.blockRadius then
                            threat = true
                            break
                        end
                    end
                end
            end
        end
        if threat ~= S.blockHeld then
            S.blockHeld = threat
            if threat then
                press(KEY_F)
            else
                release(KEY_F)
            end
        end
    elseif S.blockHeld then
        S.blockHeld = false
        release(KEY_F)
    end

    S.sweepAcc = S.sweepAcc + dt
    if S.sweepAcc >= 0.5 then
        S.sweepAcc = 0
        if Flags.instantOpen then
            sweepPrompts(true, Flags.instantPickup)
        elseif Flags.instantPickup then
            sweepPrompts(false, true)
        end
    end
end

track(RunService.Heartbeat:Connect(heartStep))

local function renderStep(dt)
    local camera = cam()

    if Flags.overrideFov then
        if camera then
            if math.abs(camera.FieldOfView - Flags.fov) > 0.1 then
                camera.FieldOfView = Flags.fov
            end
        end
    end

    refreshSilent()
    flyStep()

    local viewport = Vector2.new(1920, 1080)
    if camera then
        viewport = camera.ViewportSize
    end
    local centre = Vector2.new(viewport.X * 0.5, viewport.Y * 0.5)

    if S.ring then
        S.ring.Visible = false
        if Flags.silent then
            if Flags.showFov then
                S.ring.Visible = true
            end
        end
        S.ring.Position = centre
        local fov = 70
        if camera then
            fov = camera.FieldOfView
        end
        local halfFov = math.rad(fov) * 0.5
        if halfFov > 0 then
            local ratio = math.tan(math.rad(Flags.combatFov) * 0.5) / math.tan(halfFov)
            S.ring.Radius = ratio * (viewport.Y * 0.5)
        else
            S.ring.Radius = 0
        end
    end

    if S.marker then
        S.marker.Visible = false
        if S.silentPoint and Flags.showPoint and camera then
            local screen, onScreen = camera:WorldToViewportPoint(S.silentPoint)
            S.marker.Position = Vector2.new(screen.X, screen.Y)
            if S.silentClear then
                S.marker.Color = Color3.fromRGB(120, 240, 140)
            else
                S.marker.Color = Color3.fromRGB(240, 168, 92)
            end
            S.marker.Visible = onScreen
        end
    end

    local char = myChar()
    local tool = char and char:FindFirstChildOfClass("Tool")
    local weapon = tool and WEAPON_BY_NAME[tool.Name]
    if heldPara then
        if weapon then
            heldPara:SetValue(fmt("%s · %.1f DPS · cd %.2fs", weapon.n, weapon.dps, weapon.cd))
        else
            heldPara:SetValue("—")
        end
    end

    S.espAcc = S.espAcc + dt
    if S.espAcc >= 0.4 then
        S.espAcc = 0
        reconcileEsp()
    end
    updateEsp()

    S.stats.frames = S.stats.frames + 1
    S.stats.acc = S.stats.acc + dt
    if S.stats.acc >= 1 then
        S.stats.fps = S.stats.frames
        S.stats.frames = 0
        S.stats.acc = 0
    end
end

track(RunService.RenderStepped:Connect(renderStep))

end))

local function unload()
    Flags.autoAttack = false
    Flags.rollSell = false
    Flags.autoRun = false
    Flags.Speed = false
    Flags.Noclip = false
    Flags.silent = false
    Flags.farmEnabled = false
    S.farmRunning = false

    if S.fb.active then
        setFullBright(false)
    end
    if Flags.Fly then
        setFly(false)
    end
    if S.blockHeld then
        release(KEY_F)
    end

    for _, connection in ipairs(S.conns) do
        safeDisconnect(connection)
    end
    S.conns = {}

    for _, watcher in pairs(S.watchers) do
        safeDisconnect(watcher)
    end
    S.watchers = {}

    if S.oldNamecall then
        pcall(hookmetamethod, game, "__namecall", S.oldNamecall)
    end
    if S.oldNewIndex then
        pcall(hookmetamethod, game, "__newindex", S.oldNewIndex)
    end

    clearAllEsp()

    if S.espFolder then
        S.espFolder:Destroy()
    end
    if warnGui then
        warnGui:Destroy()
    end

    for _, fn in ipairs(S.destroy) do
        pcall(fn)
    end
    env.__StorageAdv = nil
end

env.__StorageAdv = {
    unload = unload,
    flags = Flags,
    state = S,
}
