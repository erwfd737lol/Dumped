function A.safeDisconnect(connection)
    if not connection then
        return
    end
    pcall(connection.Disconnect, connection)
end

function A.track(connection)
    table.insert(S.conns, connection)
    return connection
end

function A.onDestroy(fn)
    table.insert(S.destroy, fn)
end

function A.notify(title, content)
    pcall(function()
        WindUI:Notify({
            Title = title,
            Content = tostring(content),
            Duration = 3,
        })
    end)
    print(fmt("[Storage] %s | %s", tostring(title), tostring(content)))
end

function A.syncUI(key, value)
    local option = Options[key]
    if not option then
        return
    end
    if option.Value == value then
        return
    end
    pcall(option.Set, option, value)
end

function A.rarityRank(name)
    local tier = D.RARITY_OF[name]
    if not tier then
        return 1
    end
    return D.RARITY_RANK[tier]
end

function A.rarityColor(name)
    local tier = D.RARITY_OF[name]
    if not tier then
        return D.RARITY_COLORS.Common
    end
    return D.RARITY_COLORS[tier]
end

function A.initData()
    for _, name in ipairs(D.LEGENDARY) do
        D.RARITY_OF[name] = "Legendary"
    end
    for _, name in ipairs(D.RARE) do
        D.RARITY_OF[name] = "Rare"
    end
    for _, name in ipairs(D.UNCOMMON) do
        D.RARITY_OF[name] = "Uncommon"
    end

    for _, entry in ipairs(D.WEAPONS) do
        entry.dps = (entry.d * entry.p) / entry.cd
        D.WEAPON_BY_NAME[entry.n] = entry
    end

    table.sort(D.WEAPONS, function(a, b)
        return a.dps > b.dps
    end)

    for _, entry in ipairs(D.NPCS) do
        table.insert(D.NPC_NAMES, entry[1])
    end
end

function A.myChar()
    local live = Workspace:FindFirstChild("Live")
    if live then
        local found = live:FindFirstChild(LP.Name)
        if found then
            return found
        end
    end
    return LP.Character
end

function A.myRoot()
    local char = A.myChar()
    if not char then
        return nil
    end
    return char:FindFirstChild("HumanoidRootPart")
end

function A.myHum()
    local char = A.myChar()
    if not char then
        return nil
    end
    return char:FindFirstChildOfClass("Humanoid")
end

function A.rootOf(plr)
    local char = plr.Character
    if not char then
        return nil
    end
    return char:FindFirstChild("HumanoidRootPart")
end

function A.humOf(plr)
    local char = plr.Character
    if not char then
        return nil
    end
    return char:FindFirstChildOfClass("Humanoid")
end

function A.alive(plr)
    local hum = A.humOf(plr)
    if not hum then
        return false
    end
    if hum.Health <= 0 then
        return false
    end
    return A.rootOf(plr) ~= nil
end

function A.allRooms()
    local out = {}
    for _, folderName in ipairs(D.ROOM_FOLDERS) do
        local folder = Workspace:FindFirstChild(folderName)
        if folder then
            for _, room in ipairs(folder:GetChildren()) do
                table.insert(out, room)
            end
        end
    end
    return out
end

function A.factionOf(plr)
    local root = A.rootOf(plr)
    if not root then
        return nil
    end
    local gui = root:FindFirstChild("Faction")
    if not gui then
        return nil
    end
    local label = gui:FindFirstChild("FactionName")
    if not label then
        return nil
    end
    return label.Text
end

function A.myFaction()
    return A.factionOf(LP)
end

function A.armorOf(plr)
    local root = A.rootOf(plr)
    if not root then
        return nil
    end
    local value = root:GetAttribute("Armor")
    if type(value) ~= "number" then
        return nil
    end
    return value
end

end

function A.muzzle()
    local char = A.myChar()
    local tool = char and char:FindFirstChildOfClass("Tool")
    local handle = tool and tool:FindFirstChild("Handle")
    if handle then
        return handle.Position
    end
    local camera = A.cam()
    return camera.CFrame.Position
end

function A.partsOf(plr, mode)
    local out = {}
    local char = plr.Character
    if not char then
        return out
    end
    local order = D.PART_ORDER[mode]
    if not order then
        order = D.PART_ORDER.Head
    end
    for _, partName in ipairs(order) do
        local part = char:FindFirstChild(partName)
        if part then
            if part:IsA("BasePart") then
                table.insert(out, part)
            end
        end
    end
    return out
end

function A.fbApply()
    Lighting.Ambient = Color3.new(1, 1, 1)
    Lighting.ColorShift_Bottom = Color3.new(1, 1, 1)
    Lighting.ColorShift_Top = Color3.new(1, 1, 1)
    Lighting.Brightness = 3
    Lighting.ClockTime = 14
    Lighting.FogEnd = 1e6
    Lighting.GlobalShadows = false

    local atmosphere = Lighting:FindFirstChildOfClass("Atmosphere")
    if atmosphere then
        if atmosphere.Parent then
            if not S.fb.saved.atm then
                S.fb.saved.atm = atmosphere
            end
            atmosphere.Parent = nil
        end
    end
end

function A.setFullBright(on)
    Flags.FullBright = on
    A.syncUI("FullBright", on)

    if on then
        if S.fb.active then
            return
        end
        S.fb.active = true
        S.fb.saved = {
            ambient = Lighting.Ambient,
            shiftB = Lighting.ColorShift_Bottom,
            shiftT = Lighting.ColorShift_Top,
            brightness = Lighting.Brightness,
            clock = Lighting.ClockTime,
            fogEnd = Lighting.FogEnd,
            shadows = Lighting.GlobalShadows,
            atm = Lighting:FindFirstChildOfClass("Atmosphere"),
        }
        A.fbApply()
        S.fb.conn = Lighting.LightingChanged:Connect(function()
            if S.fb.active then
                A.fbApply()
            end
        end)
        A.notify("FullBright", "Enabled")
        return
    end

    if not S.fb.active then
        return
    end
    S.fb.active = false
    A.safeDisconnect(S.fb.conn)
    S.fb.conn = nil

    local saved = S.fb.saved
    if saved then
        Lighting.Ambient = saved.ambient
        Lighting.ColorShift_Bottom = saved.shiftB
        Lighting.ColorShift_Top = saved.shiftT
        Lighting.Brightness = saved.brightness
        Lighting.ClockTime = saved.clock
        Lighting.FogEnd = saved.fogEnd
        Lighting.GlobalShadows = saved.shadows
        if saved.atm then
            saved.atm.Parent = Lighting
        end
    end
    S.fb.saved = nil
    A.notify("FullBright", "Disabled")
end

A.onDestroy(function()
    A.safeDisconnect(S.fb.conn)
end)

function A.zeroPrompt(prompt)
    if not prompt then
        return
    end
    prompt.HoldDuration = 0
    prompt.MaxActivationDistance = math.max(prompt.MaxActivationDistance, 30)
end

function A.sweepPrompts(doors, pickup)
    for _, room in ipairs(A.allRooms()) do
        if doors then
            local door = room:FindFirstChild("Door")
            local handle = door and door:FindFirstChild("Handle")
            if handle then
                A.zeroPrompt(handle:FindFirstChild("ProximityPrompt"))
            end
        end
        if pickup then
            local contents = room:FindFirstChild("Contents")
            local loot = contents and contents:FindFirstChild("Loot")
            if loot then
                for _, item in ipairs(loot:GetChildren()) do
                    A.zeroPrompt(item:FindFirstChildOfClass("ProximityPrompt"))
                end
            end
        end
    end
end

function A.bindPromptFolders()
    for _, folderName in ipairs(D.ROOM_FOLDERS) do
        local folder = Workspace:FindFirstChild(folderName)
        if folder then
            A.track(folder.DescendantAdded:Connect(function(descendant)
                if not descendant:IsA("ProximityPrompt") then
                    return
                end
                local parent = descendant.Parent

                    A.zeroPrompt(descendant)
                    return
                end
                if not isDoor and Flags.instantPickup then
                    A.zeroPrompt(descendant)
                end
            end))
        end
    end
end

function A.nativeApply(inst, kind, on)
    local tags = D.NATIVE_TAGS[kind]
    if not tags then
        return false
    end
    local found = false
    for _, tag in ipairs(tags) do
        for _, obj in ipairs(inst:GetDescendants()) do
            if obj.Name == tag then
                local stored = S.nativeApplied[obj]
                if on then
                    if not stored then
                        stored = { enabled = obj.Enabled }
                        S.nativeApplied[obj] = stored
                    end
                    obj.Enabled = true
                    found = true
                else
                    if stored then
                        obj.Enabled = stored.enabled
                        S.nativeApplied[obj] = nil
                    end
                end
            end
        end
    end
    return found
end

function A.clearEsp(inst)
    local record = S.espObjs[inst]
    if not record then
        return
    end
    if record.hum then

    if record.gui then

    S.espObjs[inst] = nil
end

    end
end

function A.wantedEsp()
    local want = {}
    local mine = A.myFaction()

    if Flags.playersESP then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and A.alive(plr) then
                local same = false
                if mine then
                    same = A.factionOf(plr) == mine
                end
                if not (Flags.teamCheck and same) then
                    want[plr.Character] = "Player"
                end
            end
        end
    end

    if Flags.enemiesESP then
        local mobs = Workspace:FindFirstChild("Mobs")
        if mobs then
            for _, mob in ipairs(mobs:GetChildren()) do
                want[mob] = "Mob"
            end
        end
    end

    if Flags.itemsESP then
        for _, room in ipairs(A.allRooms()) do
            local contents = room:FindFirstChild("Contents")
            local loot = contents and contents:FindFirstChild("Loot")
            if loot then
                for _, item in ipairs(loot:GetChildren()) do
                    if A.rarityRank(item.Name) >= Flags.itemMinRank then
                        want[item] = "Item"
                    end
                end
            end
        end
    end

    if Flags.baseESP then
        for _, room in ipairs(A.ownRooms()) do
            want[room] = "Base"
        end
    end

    if Flags.elevatorESP then
        local elevator = Workspace:FindFirstChild("Elevator")
        if elevator then
            for _, child in ipairs(elevator:GetChildren()) do
                want[child] = "Elevator"
            end
        end
    end

    return want
end

function A.reconcileEsp()
    local want = A.wantedEsp()

    for inst in pairs(S.espObjs) do
        if not want[inst] then
            A.clearEsp(inst)
        elseif not inst.Parent then
            A.clearEsp(inst)
        end
    end

    if Flags.nativeEsp then
        for inst, kind in pairs(want) do
            local applied = A.nativeApply(inst, kind, true)
            if not applied then
                if not S.espObjs[inst] then
                    A.buildEsp(inst, kind)
                end
            elseif S.espObjs[inst] then
                A.clearEsp(inst)
            end
        end
        return
    end

    for inst, kind in pairs(want) do
        if not S.espObjs[inst] then
            A.buildEsp(inst, kind)
        end
    end
end

function A.tintFor(inst, kind)
    if kind == "Player" then
        local plr = Players:GetPlayerFromCharacter(inst)
        local faction = plr and A.factionOf(plr)
        if faction then
            local color = D.FACTION_COLORS[faction]
            if color then
                return color
            end
        end
        return Color3.fromRGB(235, 235, 235)
    end
    if kind == "Mob" then
        return Color3.fromRGB(255, 130, 110)
    end
    if kind == "Item" then
        return A.rarityColor(inst.Name)
    end
    if kind == "Base" then
        return Color3.fromRGB(120, 235, 150)
    end
    return Color3.fromRGB(180, 200, 255)
end

function A.updateEsp()
    local camera = A.cam()
    if not camera then
        return
    end
    local camPos = camera.CFrame.Position
    local viewport = camera.ViewportSize

    for inst, record in pairs(S.espObjs) do
        local pos = A.espPos(inst)
        if pos then
            local dist = (pos - camPos).Magnitude
            local near = dist <= Flags.espDist
            local tint = A.tintFor(inst, record.kind)

            record.gui.Enabled = near
            record.stroke.Color = tint
            record.name.Text = inst.Name
            record.name.TextColor3 = tint
            record.name.Visible = false
            if Flags.nameTags then
                if record.kind ~= "Item" then
                    record.name.Visible = true
                end
            end

            record.sub.Visible = record.kind == "Player"
            if record.kind == "Player" then
                local plr = Players:GetPlayerFromCharacter(inst)
                local label = ""
                if plr then
                    label = A.factionOf(plr) or ""
                    local armor = A.armorOf(plr)
                    if armor and armor > 0 then
                        label = fmt("%s · %dA", label, armor)
                    end
                end
                record.sub.Text = label
                record.sub.TextColor3 = tint
            end

            record.dist.Text = fmt("%dm", math.floor(dist))

            if record.bar then
                local hum = inst:FindFirstChildOfClass("Humanoid")
                record.barBg.Visible = false
                if hum and Flags.healthBars then
                    record.barBg.Visible = true
                    local ratio = hum.Health / math.max(hum.MaxHealth, 1)
                    ratio = math.clamp(ratio, 0, 1)
                    record.bar.Size = UDim2.fromScale(ratio, 1)
                    record.bar.BackgroundColor3 = Color3.fromRGB(
                        math.floor(255 * (1 - ratio) + 60),
                        math.floor(60 + 180 * ratio),
                        90)
                end
            end

            if record.hl then
                record.hl.Enabled = near
                record.hl.FillColor = tint
            end

            if record.tracer then
                record.tracer.Visible = false
                if Flags.tracers and near then
                    local screen, onScreen = camera:WorldToViewportPoint(pos)
                    record.tracer.Visible = onScreen
                    record.tracer.From = Vector2.new(viewport.X * 0.5, viewport.Y)
                    record.tracer.To = Vector2.new(screen.X, screen.Y)
                    record.tracer.Color = tint
                end
            end
        else
            record.gui.Enabled = false
        end
    end
end

function A.manipPoint(base, origin)
    local point = base
    if not Flags.manip then
        return point
    end
    if Flags.manipV ~= 0 then
        point = point + Vector3.yAxis * Flags.manipV
    end
    if Flags.manipL ~= 0 then
        local camera = A.cam()
        point = point + camera.CFrame.RightVector * Flags.manipL
    end
    if Flags.manipD ~= 0 then
        local delta = point - origin
        if delta.Magnitude > 0.01 then
            point = point + delta.Unit * Flags.manipD
        end
    end
    return point
end

function A.castSegment(from, to, targetChar)
    local ignore = {}
    local mine = A.myChar()
    if mine then
        table.insert(ignore, mine)
    end
    if targetChar then
        table.insert(ignore, targetChar)
    end
    S.params.FilterDescendantsInstances = ignore
    return Workspace:Raycast(from, to - from, S.params)
end

function A.solve(plr)
    local char = plr.Character
    if not char then
        return nil, false
    end

    local origin = A.muzzle()
    local best = nil
    local bestScore = -math.huge
    local clear = false

    for _, part in ipairs(A.partsOf(plr, Flags.combatPart)) do
        local point = A.manipPoint(part.Position, origin)
        local hit = A.castSegment(origin, point, char)
        local score = math.huge
        if hit then
            score = hit.Distance
        end
        if score > bestScore then
            bestScore = score
            best = point
        end
        if not hit then
            clear = true
            if score == math.huge then
                break
            end
        end
    end

    return best, clear
end

function A.passesFilter(plr)
    if plr == LP then
        return false
    end
    if not A.alive(plr) then
        return false
    end
    local mine = A.myFaction()
    if mine then
        if A.factionOf(plr) == mine then
            return false
        end
    end
    return true
end

function A.scan()
    local camera = A.cam()
    if not camera then
        return nil
    end

    local camPos = camera.CFrame.Position
    local look = camera.CFrame.LookVector
    local best = nil
    local bestScore = math.huge

    for _, plr in ipairs(Players:GetPlayers()) do
        if A.passesFilter(plr) then
            local part = A.partsOf(plr, Flags.combatPart)[1]
            if part then
                local vel = part.AssemblyLinearVelocity
                local pos = part.Position + vel * (Flags.prediction / 100)
                local delta = pos - camPos
                local dist = delta.Magnitude
                if dist > 0.5 and dist <= Flags.maxCombatDist then
                    local dot = delta.Unit:Dot(look)
                    local angle = math.deg(math.acos(math.clamp(dot, -1, 1)))
                    if angle <= Flags.combatFov then
                        if angle < bestScore then
                            bestScore = angle
                            best = { plr = plr, pos = pos }
                        end
                    end
                end
            end
        end
    end

    return best
end

function A.refreshSilent()
    if not Flags.silent then
        S.silentPoint = nil
        S.silentClear = false
        S.silentTarget = nil
        return
    end

    local target = A.scan()
    if not target then
        S.silentPoint = nil
        S.silentClear = false
        S.silentTarget = nil
        return
    end

    local point, clear = A.solve(target.plr)
    S.silentTarget = target
    S.silentClear = clear
    if point then
        S.silentPoint = point
    else
        S.silentPoint = target.pos
    end
end

function A.rewriteAimArgs(args)
    local point = S.silentPoint
    if not point then
        return nil
    end
    for index = 2, args.n do
        local valueType = typeof(args[index])
        if valueType == "CFrame" then
            args[index] = CFrame.new(A.muzzle(), point)
            return args
        end
        if valueType == "Vector3" then
            local delta = point - A.muzzle()
            if delta.Magnitude > 0.05 then
                args[index] = delta.Unit
                return args
            end
        end
    end
    return nil
end

function A.installHooks()
    if not hookmetamethod then
        return
    end

    local ok, old = pcall(function()
        return hookmetamethod(game, "__namecall", function(self, ...)
            local method = getnamecallmethod()
            if method ~= "FireServer" then
                return S.oldNamecall(self, ...)
            end
            if typeof(self) ~= "Instance" then
                return S.oldNamecall(self, ...)
            end

            if self.Name == "UnreliableRemoteEvent" then
                if Flags.spoofAim then
                    local args = table.pack(...)
                    if args[1] == "ReportAim" then
                        local rewritten = A.rewriteAimArgs(args)
                        if rewritten then
                            return S.oldNamecall(self, table.unpack(rewritten, 1, rewritten.n))
                        end
                    end
                end
                return S.oldNamecall(self, ...)
            end

            if self.Name ~= "RemoteEvent" then
                return S.oldNamecall(self, ...)
            end
            if not Flags.silent then
                return S.oldNamecall(self, ...)
            end

            local args = table.pack(...)
            local action = args[1]
            if typeof(action) ~= "string" then
                return S.oldNamecall(self, ...)
            end
            if not D.ATTACK_ACTIONS[action] then
                return S.oldNamecall(self, ...)
            end

            local rewritten = A.rewriteAimArgs(args)
            if rewritten then
                return S.oldNamecall(self, table.unpack(rewritten, 1, rewritten.n))
            end
            return S.oldNamecall(self, ...)
        end)
    end)
    if ok then
        S.oldNamecall = old
    end

    local ok2, old2 = pcall(function()
        return hookmetamethod(game, "__newindex", function(self, key, value)
            if key ~= "WalkSpeed" then
                return S.oldNewIndex(self, key, value)
            end
            if not Flags.Speed then
                return S.oldNewIndex(self, key, value)
            end
            if typeof(self) ~= "Instance" then
                return S.oldNewIndex(self, key, value)
            end
            if not self:IsA("Humanoid") then
                return S.oldNewIndex(self, key, value)
            end
            if self.Parent ~= A.myChar() then
                return S.oldNewIndex(self, key, value)
            end

            local want
            if Flags.speedMode == "Set" then
                want = Flags.speedValue
            elseif Flags.speedMode == "Multiply" then
                want = 12 * Flags.speedValue
            else
                want = 12 + Flags.speedValue
            end
            if math.abs(value - want) > 0.01 then
                value = want
            end

            return S.oldNewIndex(self, key, value)
        end)
    end)
    if ok2 then
        S.oldNewIndex = old2
        S.hookOK = true
    end
end

function A.fireAttack(action)
    local event = A.remoteEvent()
    if not event then
        return false
    end
    if Flags.requireClear then
        if S.silentTarget and not S.silentClear then
            return false
        end
    end

    local origin = A.muzzle()
    local lookAt = S.silentPoint
    if not lookAt then
        local camera = A.cam()
        lookAt = camera.CFrame.Position + camera.CFrame.LookVector * 200
    end
    if (lookAt - origin).Magnitude < 0.05 then
        lookAt = origin + Vector3.yAxis * 0.5
    end

    pcall(event.FireServer, event, action, CFrame.new(origin, lookAt))

    local releaseDelay = 0.06
    if action == "M2D" then
        releaseDelay = 0.08
    end
    task.delay(releaseDelay, function()
        local again = A.remoteEvent()
        if again then
            pcall(again.FireServer, again, action:gsub("D", "U"))
        end
    end)

    return true
end

function A.attackLoop()
    if S.attacking then
        return
    end
    S.attacking = true
    while Flags.autoAttack do
        local char = A.myChar()
        local tool = char and char:FindFirstChildOfClass("Tool")
        local weapon = tool and D.WEAPON_BY_NAME[tool.Name]
        local cooldown = 0.12
        if weapon then
            cooldown = weapon.cd
        end
        A.fireAttack("M1D")
        task.wait(math.max(cooldown, 0.05))
    end
    S.attacking = false
end

function A.setAttack(on)
    Flags.autoAttack = on
    A.syncUI("autoAttack", on)
    if on then
        task.spawn(A.attackLoop)
    end
end

    end
end

function A.flyStep()
    if not Flags.Fly then
        return
    end
    if not S.flyBV then
        return
    end
    if not S.flyBV.Parent then
        return
    end

    local camera = A.cam()
    local dir = Vector3.zero

    if UIS:IsKeyDown(Enum.KeyCode.W) then
        dir = dir + camera.CFrame.LookVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.S) then
        dir = dir - camera.CFrame.LookVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.A) then
        dir = dir - camera.CFrame.RightVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.D) then
        dir = dir + camera.CFrame.RightVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.Space) then
        dir = dir + Vector3.yAxis
    end
    if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then
        dir = dir - Vector3.yAxis
    end

    if dir.Magnitude > 0 then
        S.flyBV.Velocity = dir.Unit * Flags.flySpeed
    else
        S.flyBV.Velocity = Vector3.zero
    end
    S.flyBG.CFrame = camera.CFrame
end

function A.setCollisions(char, on)
    if not char then
        return
    end
    for _, descendant in ipairs(char:GetDescendants()) do
        if descendant:IsA("BasePart") then
            descendant.CanCollide = on
        end
    end
end

function A.farmTargets()
    local out = {}
    local names = Flags.farmNames
    local takeAll = next(names) == nil

    for _, room in ipairs(A.allRooms()) do
        local folder = room.Parent
        if folder then
            if Flags.farmScope[folder.Name] then
                local contents = room:FindFirstChild("Contents")
                local loot = contents and contents:FindFirstChild("Loot")
                if loot then
                    for _, item in ipairs(loot:GetChildren()) do
                        local wanted = takeAll
                        if not wanted then
                            local lowered = item.Name:lower()
                            for name in pairs(names) do
                                if lowered:find(name:lower(), 1, true) then
                                    wanted = true
                                    break
                                end
                            end
                        end
                        if wanted then
                            local pos = A.espPos(item)
                            if pos then
                                table.insert(out, { inst = item, pos = pos })
                            end
                        end
                    end
                end
            end
        end
    end

    return out
end

function A.stopFarm()
    S.farmRunning = false
    if S.warnFrame then
        S.warnFrame.Visible = false
    end
    pcall(CAS.UnbindAction, CAS, "AdvFarmLock")
    A.setCollisions(A.myChar(), true)
    if Flags.returnOnStop then
        if S.farmOrigin then
            local root = A.myRoot()
            if root then
                root.CFrame = S.farmOrigin
            end
        end
    end
end

function A.farmLoop()
    while S.farmRunning do
        local targets = A.farmTargets()
        if #targets == 0 then
            task.wait(0.4)
        else
            for _, target in ipairs(targets) do
                if not S.farmRunning then
                    break
                end
                local root = A.myRoot()
                if not root then
                    break
                end
                if not target.inst.Parent then
                    break
                end

                root.CFrame = CFrame.new(target.pos + Vector3.yAxis * 2.4)

                local prompt = target.inst:FindFirstChildOfClass("ProximityPrompt")
                if prompt then
                    pcall(prompt.InputHoldBegin, prompt)
                    task.delay(0.08, function()
                        pcall(prompt.InputHoldEnd, prompt)
                    end)
                end

                task.wait(Flags.farmInterval)
            end
        end
    end
    A.stopFarm()
end

function A.refreshFarm()
    if not Flags.farmEnabled then
        if S.farmRunning then
            A.stopFarm()
            A.notify("Farm", "Stopped")
        end
        return
    end

    if next(Flags.farmScope) == nil then
        if S.farmRunning then
            A.stopFarm()
        end
        return
    end

    if S.farmRunning then
        return
    end

    local root = A.myRoot()
    if not root then
        A.notify("Farm", "no character")
        return
    end
    S.farmRunning = true
    S.farmOrigin = root.CFrame
    A.setCollisions(A.myChar(), false)
    CAS:BindActionAtPriority("AdvFarmLock", function() end, false, 3000,
        Enum.PlayerActions.CharacterForward,
        Enum.PlayerActions.CharacterBackward,
        Enum.PlayerActions.CharacterLeft,
        Enum.PlayerActions.CharacterRight,
        Enum.PlayerActions.CharacterJump)
    A.showWarning()
    task.spawn(A.farmLoop)
    A.notify("Farm", "Started")
end

function A.watch(key, name, on, isMob)
    A.safeDisconnect(S.watchers[key])
    S.watchers[key] = nil
    if not on then
        return
    end

    for _, folderName in ipairs(D.ROOM_FOLDERS) do
        local folder = Workspace:FindFirstChild(folderName)
        if folder then
            S.watchers[key] = folder.DescendantAdded:Connect(function(descendant)
                if descendant.Name ~= name then
                    return
                end
                if isMob then
                    local parent = descendant.Parent
                    if parent then
                        if parent.Name == "Mobs" then
                            A.notify("Mob", name)
                        end
                    end
                    return
                end
                A.notify("NPC", name)
            end)
            return
        end
    end
end

function A.checkBounty(plr)
    if not Flags.bountyAlert then
        return
    end
    local stats = plr:FindFirstChild("leaderstats")
    local bounty = stats and stats:FindFirstChild("Bounty")
    if not bounty then
        return
    end
    if type(bounty.Value) ~= "number" then
        return
    end
    if bounty.Value >= 1000 then
        A.notify("Bounty", fmt("%s — %d", plr.Name, bounty.Value))
    end
end

function A.goBase()
    for _, room in ipairs(A.ownRooms()) do
        local floor = room:FindFirstChild("Floor")
        if floor then
            A.teleport(floor.CFrame + Vector3.new(0, 3, 0))
            A.notify("Travel", "Base")
            return
        end
    end
    A.notify("Travel", "no owned storage")
end

function A.goElevator()
    local elevator = Workspace:FindFirstChild("Elevator")
    local box = elevator and elevator:FindFirstChild("Box")
    local doors = box and box:FindFirstChild("Doors")
    if doors then
        A.teleport(doors:GetPivot())
    else
        A.teleport(D.TRAVEL_ELEVATOR)
    end
    A.notify("Travel", "Elevator")
end

function A.goNpc(name, offset)
    local char = A.myChar()
    if not char then
        return
    end
    for _, room in ipairs(A.allRooms()) do
        local contents = room:FindFirstChild("Contents")
        local folder = contents and contents:FindFirstChild("NPC")
        local npc = folder and folder:FindFirstChild(name)
        if npc then
            local pivot = npc:GetPivot()
            local behind = pivot.LookVector * offset
            char:PivotTo(CFrame.new(pivot.Position + behind + Vector3.new(0, 1.5, 0)))
            A.notify("Travel", name)
            return
        end
    end
    A.notify("Travel", name .. " not found")
end

function A.goPlayer(name)
    local plr = Players:FindFirstChild(name)
    if not plr then
        A.notify("Travel", "unavailable")
        return
    end
    if not A.alive(plr) then
        A.notify("Travel", "unavailable")
        return
    end
    local root = A.rootOf(plr)
    if root then
        A.teleport(root.CFrame + Vector3.new(0, 2, 0))
    end
end

function A.refreshPoints()
    if not U.pointDrop then
        return
    end
    local map = Workspace:FindFirstChild("Map")
    local points = map and map:FindFirstChild("Points")
    if not points then
        return
    end
    local names = {}
    for _, child in ipairs(points:GetChildren()) do
        table.insert(names, child.Name)
    end
    table.sort(names)
    pcall(U.pointDrop.SetValues, U.pointDrop, names)
end

function A.refreshWaypoints()
    if not U.wpDrop then
        return
    end
    local names = {}
    for name in pairs(S.waypoints) do
        table.insert(names, name)
    end
    table.sort(names)
    pcall(U.wpDrop.SetValues, U.wpDrop, names)
end

function A.refreshPlayers()
    if not U.plrDrop then
        return
    end
    local names = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP then
            table.insert(names, plr.Name)
        end
    end
    table.sort(names)
    pcall(U.plrDrop.SetValues, U.plrDrop, names)
end

function A.runDiagnostics()
    local lines = {}

    local function check(name, ok)
        local tag = "FAIL "
        if ok then
            tag = "PASS "
        end
        table.insert(lines, tag .. name)
    end

    check("http", game.HttpGetAsync ~= nil)
    check("drawing", Drawing ~= nil)
    check("namecall hook", S.oldNamecall ~= nil)
    check("speed hook", S.hookOK)
    check("remote event", A.remoteEvent() ~= nil)
    check("live folder", Workspace:FindFirstChild("Live") ~= nil)
    check("storages", Workspace:FindFirstChild("Storages") ~= nil)
    check("owned base", #A.ownRooms() > 0)
    check("fps over 20", S.stats.fps > 20)
    check("tabs built", S.tabsBuilt >= 8)

    if U.diagPara then
        U.diagPara:SetValue(table.concat(lines, "\n"))
    else
        print(table.concat(lines, "\n"))
    end
end

function A.onDied(char)
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then
        return
    end
    S.lastDeath = CFrame.new(root.Position)
    if U.deathPara then
        U.deathPara:SetValue(fmt("X %d, Z %d",
            math.round(root.Position.X),

    end
end

function A.heartStep(dt)
    if Flags.autoRun then
        local char = A.myChar()
        local event = A.remoteEvent()
        if char then
            char:SetAttribute("Running", true)
        end
        if event then
            pcall(event.FireServer, event, "Run")
        end
    end

    if Flags.Noclip then
        local char = A.myChar()
        if char then
            for _, descendant in ipairs(char:GetDescendants()) do
                if descendant:IsA("BasePart") then
                    if descendant.CanCollide then
                        descendant.CanCollide = false
                    end
                end
            end
        end
    end

    if Flags.Speed then
        local hum = A.myHum()
        if hum then
            local want
            if Flags.speedMode == "Set" then
                want = Flags.speedValue
            elseif Flags.speedMode == "Multiply" then
                want = 12 * Flags.speedValue
            else
                want = 12 + Flags.speedValue
            end
            if math.abs(hum.WalkSpeed - want) > 0.01 then
                hum.WalkSpeed = want
            end
        end
    end

    if Flags.autoReload then
        local char = A.myChar()
        local tool = char and char:FindFirstChildOfClass("Tool")
        local event = A.remoteEvent()
        if tool and event then
            if D.WEAPON_BY_NAME[tool.Name] then
                pcall(event.FireServer, event, "Reload")
            end
        end
    end

    if Flags.autoKnife then
        A.knifeIfEmpty()
    end

    if Flags.autoBlock then
        local root = A.myRoot()
        local threat = false
        if root then
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LP and A.alive(plr) then
                    local other = A.rootOf(plr)
                    if other then
                        if (other.Position - root.Position).Magnitude <= Flags.blockRadius then
                            threat = true
                            break
                        end
                    end
                end
            end
        end
        if threat ~= S.blockHeld then
            S.blockHeld = threat
            if threat then
                A.press(D.KEY_F)
            else
                A.release(D.KEY_F)
            end
        end
    elseif S.blockHeld then
        S.blockHeld = false
        A.release(D.KEY_F)
    end

    S.sweepAcc = S.sweepAcc + dt
    if S.sweepAcc >= 0.5 then
        S.sweepAcc = 0
        if Flags.instantOpen then
            A.sweepPrompts(true, Flags.instantPickup)
        elseif Flags.instantPickup then
            A.sweepPrompts(false, true)
        end
    end
end

function A.renderStep(dt)
    local camera = A.cam()

    if Flags.overrideFov then
        if camera then
            if math.abs(camera.FieldOfView - Flags.fov) > 0.1 then
                camera.FieldOfView = Flags.fov
            end
        end
    end

    A.refreshSilent()
    A.flyStep()

    local viewport = Vector2.new(1920, 1080)
    if camera then
        viewport = camera.ViewportSize
    end
    local centre = Vector2.new(viewport.X * 0.5, viewport.Y * 0.5)

    if S.ring then
        S.ring.Visible = false
        if Flags.silent then
            if Flags.showFov then
                S.ring.Visible = true
            end
        end
        S.ring.Position = centre
        local fov = 70
        if camera then
            fov = camera.FieldOfView
        end
        local halfFov = math.rad(fov) * 0.5
        if halfFov > 0 then
            local ratio = math.tan(math.rad(Flags.combatFov) * 0.5) / math.tan(halfFov)
            S.ring.Radius = ratio * (viewport.Y * 0.5)
        else
            S.ring.Radius = 0
        end
    end

    if S.marker then
        S.marker.Visible = false
        if S.silentPoint and Flags.showPoint and camera then
            local screen, onScreen = camera:WorldToViewportPoint(S.silentPoint)
            S.marker.Position = Vector2.new(screen.X, screen.Y)
            if S.silentClear then
                S.marker.Color = Color3.fromRGB(120, 240, 140)
            else
                S.marker.Color = Color3.fromRGB(240, 168, 92)
            end
            S.marker.Visible = onScreen
        end
    end

    if U.heldPara then
        local char = A.myChar()
        local tool = char and char:FindFirstChildOfClass("Tool")
        local weapon = tool and D.WEAPON_BY_NAME[tool.Name]
        if weapon then
            U.heldPara:SetValue(fmt("%s · %.1f DPS · cd %.2fs", weapon.n, weapon.dps, weapon.cd))
        else
            U.heldPara:SetValue("—")
        end
    end

    S.espAcc = S.espAcc + dt
    if S.espAcc >= 0.4 then
        S.espAcc = 0
        A.reconcileEsp()
    end
    A.updateEsp()

    S.stats.frames = S.stats.frames + 1
    S.stats.acc = S.stats.acc + dt
    if S.stats.acc >= 1 then
        S.stats.fps = S.stats.frames
        S.stats.frames = 0
        S.stats.acc = 0
    end
end

    A.track(RunService.RenderStepped:Connect(A.renderStep))
    A.track(Workspace.ChildAdded:Connect(function()
        task.delay(0.5, A.reconcileEsp)
    end))

    A.refreshPoints()
    A.refreshPlayers()
    Window:SelectTab(1)
    A.notify("The Storage", fmt("ready · %d tabs built", S.tabsBuilt))
end

--═══════════════════════════════════════════════════════════════
--  TEARDOWN
--═══════════════════════════════════════════════════════════════
function A.unload()
    Flags.autoAttack = false
    Flags.rollSell = false
    Flags.autoRun = false
    Flags.Speed = false
    Flags.Noclip = false
    Flags.silent = false
    Flags.farmEnabled = false
    S.farmRunning = false

    if S.fb.active then
        A.setFullBright(false)
    end
    if Flags.Fly then
        A.setFly(false)
    end
    if S.blockHeld then
        A.release(D.KEY_F)
    end

    for _, connection in ipairs(S.conns) do
        A.safeDisconnect(connection)
    end
    S.conns = {}

    for _, watcher in pairs(S.watchers) do
        A.safeDisconnect(watcher)
    end
    S.watchers = {}

    if S.oldNamecall then
        pcall(hookmetamethod, game, "__namecall", S.oldNamecall)
    end
    if S.oldNewIndex then
        pcall(hookmetamethod, game, "__newindex", S.oldNewIndex)
    end

    A.clearAllEsp()

    if S.espFolder then
        S.espFolder:Destroy()
    end
    if S.warnGui then
        S.warnGui:Destroy()
    end

    for _, fn in ipairs(S.destroy) do
        pcall(fn)
    end
    env.__StorageAdv = nil
end

env.__StorageAdv = {
    unload = A.unload,
    flags = Flags,
    state = S,
}

A.boot()
