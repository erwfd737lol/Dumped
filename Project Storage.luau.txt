function Env.get(name)
    local getter = LEX[name]
    if getter then
        local ok, v = pcall(getter)
        if ok and v ~= nil then return v end
    end
    for _, source in ipairs({ function() return getgenv()[name] end, function() return _G[name] end,
        function() return rawget(getfenv(0), name) end }) do
        local ok, v = pcall(source)
        if ok and v ~= nil then return v end
    end
    return nil
end

local function shim(name, fallback)
    local v = Env.get(name)
    if v ~= nil then return v end
    return fallback
end

function Env.probeDrawing()
    if CAP.drawing ~= nil then return CAP.drawing end
    local d = Env.get("Drawing")
    CAP.drawing = d ~= nil and pcall(function()
        local o = d.new("Square")
        o.Visible = false
        o:Remove()
    end) or false
    return CAP.drawing
end

function Env.probeHook()
    if CAP.hook ~= nil then return CAP.hook end
    local ok = hookmetamethod ~= nil and getnamecallmethod ~= nil
    if ok then
        ok = pcall(function()
            local o = hookmetamethod(game, "__namecall", newcclosure(function() end))
            hookmetamethod(game, "__namecall", o)
        end)
    end
    CAP.hook = ok
    return ok
end

Util.tracked = {}

function Util.track(c)
    Util.tracked[#Util.tracked + 1] = c
    return c
end

function Util.untrackAll()
    for _, c in ipairs(Util.tracked) do pcall(function() c:Disconnect() end) end
    table.clear(Util.tracked)
end

function Util.character()
    local c = LocalPlayer.Character
    return (c and c.Parent) and c or nil
end

function Util.humanoid()
    local c = Util.character()
    return c and c:FindFirstChildOfClass("Humanoid")
end

function Util.root()
    local c = Util.character()
    return c and c:FindFirstChild("HumanoidRootPart")
end

function Util.alive()
    local h = Util.humanoid()
    return h ~= nil and h.Health > 0
end

function Util.cam()
    return Workspace.CurrentCamera
end

function Util.screen(pos)
    local cam = Util.cam()
    if not cam then return nil end
    local p = cam:WorldToViewportPoint(pos)
    if p.Z <= 0 then return nil end
    return Vector2.new(p.X, p.Y)
end

function Util.raycast(origin, dir, ignore)
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = ignore or { Util.character() }
    rp.IgnoreWater = true
    local ok, r = pcall(Workspace.Raycast, Workspace, origin, dir, rp)
    return ok and r or nil
end

function Util.visible(a, b)
    local d = b - a
    local hit = Util.raycast(a, d)
    return not hit or (hit.Position - a).Magnitude >= d.Magnitude - 1.5
end

function Util.container(n)
    return Workspace:FindFirstChild(n)
end

function Util.part(i)
    if i:IsA("BasePart") then return i end
    return i:FindFirstChildWhichIsA("BasePart", true)
end

local roomCache, roomAt = {}, 0
function Util.rooms()
    local now = tick()
    if now - roomAt < 1 and #roomCache > 0 then return roomCache end
    roomAt = now
    local out = {}
    for _, n in ipairs({ SURFACE, BASEMENT }) do
        local c = Util.container(n)
        if c then
            for _, r in ipairs(c:GetChildren()) do out[#out + 1] = r end
        end
    end
    roomCache = out
    return out
end

function Util.dist(a, b)
    return (a - b).Magnitude
end

Perf = { frames = 0, acc = 0, fps = 0 }

function Perf.sample(dt)
    Perf.frames += 1
    Perf.acc += dt
    if Perf.acc >= 1 then
        Perf.fps = math.floor(Perf.frames / Perf.acc + 0.5)
        Perf.frames, Perf.acc = 0, 0
    end
end

Sched = { tasks = {}, conn = nil }

function Sched.add(name, interval, fn)
    Sched.tasks[name] = { interval = interval, nextAt = 0, fn = fn, enabled = false }
end

function Sched.on(name, enabled)
    local t = Sched.tasks[name]
    if t then t.enabled = enabled end
end

function Sched.start()
    if Sched.conn then return end
    Sched.conn = RunService.Heartbeat:Connect(function()
        local now = tick()
        for _, t in pairs(Sched.tasks) do
            if t.enabled and now >= t.nextAt then
                t.nextAt = now + t.interval
                local ok, err = pcall(t.fn)
                if not ok then warn("[SS] task " .. err) end
            end
        end
    end)
end

function Sched.stop()
    if Sched.conn then
        pcall(function() Sched.conn:Disconnect() end)
        Sched.conn = nil
    end
end

Binds = { list = {}, capturing = nil }

function Binds.add(action, get, run)
    Binds.list[#Binds.list + 1] = { action = action, get = get, run = run }
end

function Binds.dispatch(code)
    for _, b in ipairs(Binds.list) do
        if b.get() == code then
            b.run()
            return true
        end
    end
    return false
end

Cfg = { backend = "none", dirty = false }

function Cfg.detect()
    local w, r = Env.get("writefile"), Env.get("readfile")
    if w and r and pcall(w, PATH, "{}") then Cfg.backend = "file"
    elseif getgenv then Cfg.backend = "memory"
    else Cfg.backend = "none" end
end

function Cfg.merge(section, values)
    local target = Config[section]
    if type(target) ~= "table" then return end
    for k, def in pairs(target) do
        local v = values[k]
        if type(def) == type(v) and type(v) ~= "table" then target[k] = v
        elseif type(def) == "table" and type(v) == "table" then
            if k == "Colors" then
                for n, c in pairs(v) do
                    if target.Colors[n] and typeof(c) == "Color3" then target.Colors[n] = c end
                end
            else
                for n, c in pairs(v) do target[n] = c end
            end
        end
    end
end

function Cfg.serialize()
    local out = {}
    for s, sec in pairs(Config) do
        local copy = {}
        for k, v in pairs(sec) do
            if k == "Colors" then
                copy.Colors = {}
                for n, c in pairs(v) do
                    copy.Colors[n] = { math.floor(c.R * 255), math.floor(c.G * 255), math.floor(c.B * 255) }
                end
            elseif type(v) ~= "table" then
                copy[k] = typeof(v) == "EnumItem" and v.Name or v
            else
                copy[k] = v
            end
        end
        out[s] = copy
    end
    return out
end

function Cfg.save()
    if Cfg.backend == "none" or Cfg.dirty then return end
    Cfg.dirty = true
    task.delay(1, function()
        Cfg.dirty = false
        local ok, enc = pcall(HttpService.JSONEncode, HttpService, Cfg.serialize())
        if not ok then return end
        if Cfg.backend == "file" then
            local w = Env.get("writefile")
            if w then pcall(w, PATH, enc) end
        else
            pcall(function() getgenv()[MEM] = enc end)
        end
    end)
end

function Cfg.load()
    if Cfg.backend == "none" then return end
    local raw
    if Cfg.backend == "file" then
        local isf, rd = Env.get("isfile"), Env.get("readfile")
        if isf and not isf(PATH) then return end
        if rd then
            local ok, c = pcall(rd, PATH)
            if ok then raw = c end
        end
    else
        local ok, c = pcall(function() return getgenv()[MEM] end)
        if ok then raw = c end
    end
    if type(raw) ~= "string" or #raw < 2 then return end
    local ok, dec = pcall(HttpService.JSONDecode, HttpService, raw)
    if not ok or type(dec) ~= "table" then return end
    for s, v in pairs(dec) do
        if type(v) == "table" and Config[s] then
            if v.Colors then
                for n, c in pairs(v.Colors) do
                    if type(c) == "table" and Config[s].Colors[n] then
                        v.Colors[n] = rgb(c[1] or 255, c[2] or 255, c[3] or 255)
                    end
                end
            end
            for k, val in pairs(v) do
                if k ~= "Colors" and type(val) == "string" then
                    local eo, ei = pcall(function() return Enum.KeyCode[val] end)
                    if eo and ei and type(Config[s][k]) == "EnumItem" then v[k] = ei end
                end
            end
            Cfg.merge(s, v)
        end
    end
end

function Cfg.export()
    local ok, enc = pcall(HttpService.JSONEncode, HttpService, Cfg.serialize())
    if not ok then return nil end
    if setclipboard then pcall(setclipboard, enc) end
    return enc
end

function Cfg.import(text)
    local ok, dec = pcall(HttpService.JSONDecode, HttpService, text)
    if not ok or type(dec) ~= "table" then return false end
    for s, v in pairs(dec) do
        if type(v) == "table" and Config[s] then Cfg.merge(s, v) end
    end
    Cfg.save()
    return true
end

function Diagnostics.run()
    table.clear(Diagnostics.results)
    local _, v = identifyexecutor()

    Diagnostics.add("FPS", Perf.fps >= 30, tostring(Perf.fps))
    Diagnostics.add("Drawing", HAS_DRAWING, HAS_DRAWING and "available" or "overlay limited")

    Diagnostics.add("Backend", Cfg.backend ~= "none", Cfg.backend)
    Diagnostics.add("Clipboard", setclipboard ~= nil, setclipboard and "available" or "unavailable")
    Diagnostics.add("ESP index", true, tostring(NativeESP.count()))
    Diagnostics.add("Overlay", true, tostring(OverlayESP.count()))

    Refs.refresh()
    Diagnostics.add("Remotes", Refs.RE ~= nil)
    Diagnostics.add("Live folder", Util.container(LIVE) ~= nil)

    local room = Panic.ownedRoom()
    Diagnostics.add("Owned base", room ~= nil, room and room.Name or "none")
    Diagnostics.add("Bed", room ~= nil and Panic.bedIn(room) ~= nil)

    local rep = Movement.report()
    Diagnostics.add("Speed writer", rep.writer, rep.detail)
    Diagnostics.add("Override", rep.holds, rep.delta)

    for _, r in ipairs(Diagnostics.results) do
        print(("[SS] %s %s%s"):format(r.pass and "PASS" or "FAIL", r.label, r.detail and (" - " .. r.detail) or ""))
    end
end

NativeESP = { index = {}, conns = {}, pending = false }

local TAGS = { Player = { "PlayerHighlight", "PlayerESP" }, Mob = { "MobHighlight", "MobESP" },
    Item = { "LootHighlight", "LootName" } }

local function pushObj(entry, o)
    if o:IsA("Highlight") or o:IsA("BillboardGui") or o:IsA("SurfaceGui")
        or o:IsA("ScreenGui") or o:IsA("Beam") then
        entry.objects[#entry.objects + 1] = o
    end
end

local function indexModel(model, kind, tags)
    local entry = { model = model, kind = kind, objects = {}, last = nil }
    for _, t in ipairs(tags or {}) do
        local f = model:FindFirstChild(t, true)
        if f then pushObj(entry, f) end
    end
    local folder = model:FindFirstChild("ESP")
    if folder then
        for _, d in ipairs(folder:GetDescendants()) do pushObj(entry, d) end
    end
    if #entry.objects > 0 then NativeESP.index[#NativeESP.index + 1] = entry end
end

function NativeESP.count() return #NativeESP.index end

function NativeESP.rebuild()
    table.clear(NativeESP.index)
    local live = Util.container(LIVE)
    if live then
        for _, m in ipairs(live:GetChildren()) do
            if m ~= LocalPlayer.Character and m:IsA("Model") then indexModel(m, "Player", TAGS.Player) end
        end
    end
    local mobs = Util.container(MOBS)
    if mobs then
        for _, m in ipairs(mobs:GetChildren()) do
            if m:IsA("Model") then indexModel(m, "Mob", TAGS.Mob) end
        end
    end
    for _, room in ipairs(Util.rooms()) do
        local contents = room:FindFirstChild("Contents")
        if contents then
            local loot = contents:FindFirstChild("Loot")
            if loot then
                for _, i in ipairs(loot:GetChildren()) do indexModel(i, "Item", TAGS.Item) end
            end
            local rm = contents:FindFirstChild("Mobs")
            if rm then
                for _, m in ipairs(rm:GetChildren()) do
                    if m:IsA("Model") then indexModel(m, "Mob", TAGS.Mob) end
                end
            end
        end
        if Panic.isOwned(room) then indexModel(room, "Base", { "ESP" }) end
    end
    local ev = Util.container("Elevator")
    if ev then
        for _, c in ipairs(ev:GetChildren()) do indexModel(c, "Elevator", { "ESP" }) end
    end
end

function NativeESP.queueRebuild()
    if NativeESP.pending then return end
    NativeESP.pending = true
    task.delay(0.6, function()
        NativeESP.pending = false
        NativeESP.rebuild()
    end)
end

local function allowed(entry)
    if entry.kind == "Player" then
        if not Config.Native.Players then return false end
        local filter = Config.ESP.PlayerFilter
        if filter ~= "" and not entry.model.Name:lower():find(filter:lower(), 1, true) then return false end
        return true
    elseif entry.kind == "Mob" then
        if not Config.Native.Mobs then return false end
        local filter = Config.ESP.MobFilter
        if filter ~= "" and not entry.model.Name:lower():find(filter:lower(), 1, true) then return false end
        return true
    elseif entry.kind == "Item" then
        return Config.Native.Items
    end
    return entry.kind == "Base" and Config.Native.Base or entry.kind == "Elevator" and Config.Native.Elevator
end

function NativeESP.cull()
    if #NativeESP.index == 0 then return end
    local cam = Util.cam()
    local origin = cam and cam.CFrame.Position
    local budgets = { Player = Config.ESP.MaxPlayers, Mob = Config.ESP.MaxMobs, Item = Config.ESP.MaxItems }
    local used = { Player = 0, Mob = 0, Item = 0 }

    for _, e in ipairs(NativeESP.index) do
        local want = false
        if origin and allowed(e) then
            local distLimit = e.kind == "Player" and Config.ESP.DistPlayers
                or e.kind == "Mob" and Config.ESP.DistMobs
                or e.kind == "Item" and Config.ESP.DistItems or math.huge
            if e.kind == "Base" or e.kind == "Elevator" then
                want = true
            else
                local a = e.anchor
                if not (a and a.Parent) then
                    a = Util.part(e.model)
                    e.anchor = a
                end
                if a then
                    want = Util.dist(a.Position, origin) <= distLimit
                    if want and budgets[e.kind] then
                        used[e.kind] += 1
                        if used[e.kind] > budgets[e.kind] then want = false end
                    end
                end
            end
        end
        if want ~= e.last then
            e.last = want
            for _, o in ipairs(e.objects) do
                if o.Parent then pcall(function() o.Enabled = want end) end
            end
        end
    end
end

function NativeESP.applyTags()
    if not Config.ESP.HideVanilla then return end
    local live = Util.container(LIVE)
    if not live then return end
    for _, m in ipairs(live:GetChildren()) do
        local h = m:FindFirstChildOfClass("Humanoid")
        if h then
            local want = Config.Native.Players and Enum.HumanoidDisplayDistanceType.None
                or Enum.HumanoidDisplayDistanceType.Subject
            if h.DisplayDistanceType ~= want then h.DisplayDistanceType = want end
        end
    end
end

function NativeESP.start()
    NativeESP.stop()
    NativeESP.rebuild()
    for _, n in ipairs({ LIVE, MOBS, SURFACE, BASEMENT }) do
        local f = Util.container(n)
        if f then
            NativeESP.conns[#NativeESP.conns + 1] = f.ChildAdded:Connect(NativeESP.queueRebuild)
            NativeESP.conns[#NativeESP.conns + 1] = f.ChildRemoved:Connect(NativeESP.queueRebuild)
        end
    end
    Sched.on("cull", true)
    Sched.on("tags", true)
end

function NativeESP.stop()
    for _, c in ipairs(NativeESP.conns) do pcall(function() c:Disconnect() end) end
    table.clear(NativeESP.conns)
    Sched.on("cull", false)
    Sched.on("tags", false)
    table.clear(NativeESP.index)
end

OverlayESP = { entries = {} }

local function dispose(e)
    if e.highlight then pcall(function() e.highlight:Destroy() end) end
    if e.gui then pcall(function() e.gui:Destroy() end) end
end

local function build(instance, kind)
    local e = { instance = instance, kind = kind, anchor = nil }
    if not Config.ESP.HighlightOnly then
        local hl = Instance.new("Highlight")
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 0.72
        hl.Parent = instance
        e.highlight = hl
    end
    if not Config.ESP.HighlightOnly then
        local gui = Instance.new("BillboardGui")
        gui.Size = UDim2.fromScale(12, 2)
        gui.StudsOffsetWorldSpace = Vector3.new(0, 2.9, 0)
        gui.AlwaysOnTop = true
        gui.MaxDistance = Config.Perf.OvDist
        local name = Instance.new("TextLabel")
        name.Size = UDim2.new(1, 0, 0.5, 0)
        name.BackgroundTransparency = 1
        name.Font = Enum.Font.GothamBold
        name.TextSize = 13
        name.TextStrokeTransparency = 0.35
        name.Parent = gui
        local sub = Instance.new("TextLabel")
        sub.Size = UDim2.new(1, 0, 0.5, 0)
        sub.Position = UDim2.new(0, 0, 0.5, 0)
        sub.BackgroundTransparency = 1
        sub.Font = Enum.Font.Gotham
        sub.TextSize = 12
        sub.TextStrokeTransparency = 0.35
        sub.Parent = gui
        gui.Parent = instance
        e.gui, e.name, e.sub = gui, name, sub
    end
    return e
end

function OverlayESP.count()
    local n = 0
    for _ in pairs(OverlayESP.entries) do n += 1 end
    return n
end

function OverlayESP.activeFor(kind)
    if Config.ESP.Mode == "Off" or Config.ESP.Mode == "Native" then return false end
    if kind == "Player" then return Config.Overlay.Players end
    if kind == "Mob" then return Config.Overlay.Mobs end
    if kind == "Item" then return Config.Overlay.Items end
    return false
end

function OverlayESP.onFound(instance, kind)
    if OverlayESP.entries[instance] or not OverlayESP.activeFor(kind) then return end
    if OverlayESP.count() >= Config.Perf.OvMax then return end
    OverlayESP.entries[instance] = build(instance, kind)
end

function OverlayESP.onLost(instance)
    local e = OverlayESP.entries[instance]
    if not e then return end
    dispose(e)
    OverlayESP.entries[instance] = nil
end

function OverlayESP.resync()
    for i in pairs(OverlayESP.entries) do
        if not i.Parent or not Scanner.known[i] then OverlayESP.onLost(i) end
    end
    for i, k in pairs(Scanner.known) do
        if OverlayESP.activeFor(k) then OverlayESP.onFound(i, k) end
    end
end

function OverlayESP.update()
    if Config.ESP.Mode == "Off" or Config.ESP.Mode == "Native" then return end
    local cam = Util.cam()
    if not cam then return end
    local origin = cam.CFrame.Position
    local center = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)

    for i, e in pairs(OverlayESP.entries) do
        if not i.Parent then
            OverlayESP.onLost(i)
            continue
        end
        local a = e.anchor
        if not (a and a.Parent) then
            a = Util.part(i)
            e.anchor = a
        end
        local show = a ~= nil
        local d = 0
        if show then
            d = Util.dist(a.Position, origin)
            if d > Config.Perf.OvDist then show = false end
        end
        if show and e.kind == "Player" and e.player then
            if Config.ESP.TeamCheck and e.player.Team == LocalPlayer.Team then show = false end
            if Config.ESP.FriendCheck and e.friend then show = false end
        end
        if not show then
            if e.gui and e.gui.Enabled then e.gui.Enabled = false end
            if e.highlight then e.highlight.Enabled = false end
            continue
        end

        local color = e.kind == "Item"
            and (Config.ESP.RarityColor and Config.Overlay.Colors[rarityOf(i.Name)] or Config.Overlay.Colors.Common)
            or e.kind == "Mob" and Config.Overlay.Colors.Mob
            or e.team and Config.Overlay.Colors.Team
            or e.friend and Config.Overlay.Colors.Friend
            or Config.Overlay.Colors.Player

        if e.highlight then
            e.highlight.Enabled = true
            e.highlight.FillColor = color
            e.highlight.OutlineColor = color
        end
        if e.gui then
            e.gui.Enabled = true
            e.name.Text = Config.Overlay.Name and i.Name or ""
            e.name.TextColor3 = color
            e.sub.Text = Config.Overlay.Distance and ("%d m"):format(d) or ""
            e.sub.TextColor3 = color
            e.gui.StudsOffsetWorldSpace = Config.Overlay.Health
                and Vector3.new(0, 2.9, 0) or Vector3.new(0, 2.9, 0)
        end
        if Config.Overlay.Tracer then
            local s = Util.screen(a.Position)
            if s then e.screenPos = s end
        end
        e.centerOffset = e.gui and Util.screen(a.Position) and (Util.screen(a.Position) - center).Magnitude or nil
    end
end

function OverlayESP.stop()
    for i in pairs(OverlayESP.entries) do OverlayESP.onLost(i) end
end

Scanner = { known = {} }

function Scanner.register(i, k)
    if Scanner.known[i] then return end
    Scanner.known[i] = k
    OverlayESP.onFound(i, k)
end

    if not Scanner.known[i] then return end

end

function Scanner.sweep()
    if Config.ESP.Mode == "Off" or Config.ESP.Mode == "Native" then return end
    local live = Util.container(LIVE)
    if live and Config.Overlay.Players then
        for _, m in ipairs(live:GetChildren()) do
            if m:IsA("Model") and m ~= LocalPlayer.Character then Scanner.register(m, "Player") end
        end
    end
    local mobs = Util.container(MOBS)
    if mobs and Config.Overlay.Mobs then
        for _, m in ipairs(mobs:GetChildren()) do
            if m:IsA("Model") then Scanner.register(m, "Mob") end
        end
    end
    if Config.Overlay.Items then
        for _, r in ipairs(Util.rooms()) do
            local c = r:FindFirstChild("Contents")
            local loot = c and c:FindFirstChild("Loot")
            if loot then
                for _, i in ipairs(loot:GetChildren()) do Scanner.register(i, "Item") end
            end
        end
    end
end

    for i in pairs(Scanner.known) do OverlayESP.onLost(i) end
    table.clear(Scanner.known)
end

Movement = {
    hook = nil, writing = false, suspended = false, writerOk = false,
    base = nil, flyV = nil, flyG = nil, noclipConn = nil, jumpConn = nil, bhopConn = nil,
    runState = false, current = nil,
}

local DEFAULT_SPEED = 16

function Movement.isTarget(h)
    return h:IsA("Humanoid") and h.Parent ~= nil and h.Parent == LocalPlayer.Character
end

function Movement.write(h, k, v)
    Movement.writing = true
    local ok = pcall(function() h[k] = v end)
    Movement.writing = false
    return ok
end

function Movement.baseSpeed()
    local b = Movement.base
    if b and b.h == Util.humanoid() then return b.speed end
    return DEFAULT_SPEED
end

function Movement.targetSpeed()
    if Movement.suspended or not State.master or not Config.Player.Speed then return nil end
    local v = Config.Player.SpeedValue
    if Config.Player.SpeedMode == "Add" then return Movement.baseSpeed() + v end
    if Config.Player.SpeedMode == "Multiply" then return Movement.baseSpeed() * (v / 10) end
    return v
end

function Movement.installHook()
    if Movement.hook or not HAS_HOOKS then return end
    pcall(function()
        Movement.hook = hookmetamethod(game, "__newindex", newcclosure(function(self, k, v)
            if Movement.writing or k ~= "WalkSpeed" then return Movement.hook(self, k, v) end
            if not Movement.isTarget(self) then return Movement.hook(self, k, v) end
            local t = Movement.targetSpeed()
            if t then v = t end
            return Movement.hook(self, k, v)
        end))
    end)
end

function Movement.uninstallHook()
    if not Movement.hook then return end
    pcall(function() hookmetamethod(game, "__newindex", Movement.hook) end)
    Movement.hook = nil
end

local function writerStep(dt)
    if Movement.suspended or not State.master then return end
    local h = Util.humanoid()
    if not h then return end
    local t = Movement.targetSpeed()
    if not t then return end

    local cur = Movement.current or h.WalkSpeed
    local ramp = Config.Player.SpeedRamp
    if ramp > 0 and math.abs(t - cur) > 0.05 then
        local step = ramp * math.max(dt, 0.001)
        cur = t > cur and math.min(t, cur + step) or math.max(t, cur - step)
    else
        cur = t
    end
    Movement.current = cur
    if math.abs(h.WalkSpeed - cur) > 0.01 then Movement.write(h, "WalkSpeed", cur) end
end

function Movement.bindWriter()
    if Movement.writerOk then return end
    Movement.writerOk = pcall(function()
        RunService:UnbindFromRenderStep(WRITER_STEP)
        RunService:BindToRenderStep(WRITER_STEP, Enum.RenderPriority.Last.Value, writerStep)
    end)
end

function Movement.unbindWriter()
    if not Movement.writerOk then return end
    pcall(function() RunService:UnbindFromRenderStep(WRITER_STEP) end)
    Movement.writerOk = false
end

function Movement.capture(h)
    if Movement.base and Movement.base.h == h then return end
    local cur = h.WalkSpeed
    Movement.base = { h = h, speed = cur > 0 and cur or DEFAULT_SPEED }
end

function Movement.applySpeed()
    local h = Util.humanoid()
    if not h then return end
    Movement.capture(h)
    Movement.bindWriter()
    Movement.current = nil
    local t = Movement.targetSpeed()
    if t and Config.Player.SpeedRamp <= 0 then Movement.write(h, "WalkSpeed", t) end
end

function Movement.revertSpeed()
    local h = Util.humanoid()
    local b = Movement.base
    Movement.current = nil
    if h and b and b.h == h then Movement.write(h, "WalkSpeed", b.speed) end
end

function Movement.onSpeed(on)
    if on then
        Movement.suspended = false
        Movement.applySpeed()
    else
        Movement.revertSpeed()
    end
end

function Movement.onSpeedValue()
    if not Config.Player.Speed then return end
    Movement.current = nil
    if Config.Player.SpeedRamp <= 0 then
        local h = Util.humanoid()
        local t = Movement.targetSpeed()
        if h and t then Movement.write(h, "WalkSpeed", t) end
    end
end

function Movement.setNoclip(on)
    if Movement.noclipConn then
        pcall(function() Movement.noclipConn:Disconnect() end)
        Movement.noclipConn = nil
    end
    local c = Util.character()
    if not on then
        if c then
            for _, p in ipairs(c:GetDescendants()) do
                if p:IsA("BasePart") then p.CanCollide = true end
            end
        end
        return
    end
    Movement.noclipConn = RunService.Stepped:Connect(function()
        local ch = Util.character()
        if not ch then return end
        for _, p in ipairs(ch:GetDescendants()) do
            if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
        end
    end)
end

function Movement.setInfJump(on)
    if Movement.jumpConn then
        pcall(function() Movement.jumpConn:Disconnect() end)
        Movement.jumpConn = nil
    end
    if not on then return end
    Movement.jumpConn = UserInputService.JumpRequest:Connect(function()
        local h = Util.humanoid()
        if h then pcall(function() h:ChangeState(Enum.HumanoidStateType.Jumping) end) end
    end)
end

function Movement.setBhop(on)
    if Movement.bhopConn then
        pcall(function() Movement.bhopConn:Disconnect() end)
        Movement.bhopConn = nil
    end
    if not on then return end
    local h = Util.humanoid()
    if not h then return end
    Movement.bhopConn = h.StateChanged:Connect(function(_, new)
        if new ~= Enum.HumanoidStateType.Landed then return end
        local hum = Util.humanoid()
        if not hum then return end
        if hum.MoveDirection.Magnitude > 0.1 then
            pcall(function() hum:ChangeState(Enum.HumanoidStateType.Jumping) end)
        end
    end)
end

function Movement.setAutoRun(on)
    if on == Movement.runState then return end
    Movement.runState = on
    Refs.fire("Run")
end

function Movement.startFly()
    local r, h = Util.root(), Util.humanoid()
    if not r or not h then return end
    Movement.stopFly()
    h.PlatformStand = true
    local v = Instance.new("BodyVelocity")
    v.MaxForce = Vector3.new(4e4, 4e4, 4e4)
    v.Parent = r
    Movement.flyV = v
    local g = Instance.new("BodyGyro")
    g.MaxTorque = Vector3.new(4e4, 4e4, 4e4)
    g.P = 2000
    g.D = 200
    g.Parent = r
    Movement.flyG = g
end

function Movement.stopFly()
    if Movement.flyV then
        pcall(function() Movement.flyV:Destroy() end)
        Movement.flyV = nil
    end
    if Movement.flyG then
        pcall(function() Movement.flyG:Destroy() end)
        Movement.flyG = nil
    end
    local h = Util.humanoid()
    if h then h.PlatformStand = false end
end

function Movement.updateFly()
    if not Movement.flyV or not Movement.flyG then return end
    local r, cam = Util.root(), Util.cam()
    if not r or not cam then return end

    if Config.Player.FlyHold and not State.flyHeld then
        Movement.flyV.Velocity = Vector3.zero
        Movement.flyG.CFrame = CFrame.new(r.Position) * cam.CFrame.Rotation
        return
    end

    local dir = Vector3.zero
    local mode = Config.Player.FlyMode
    if mode ~= "Vertical" then
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cam.CFrame.RightVector end
    end
    if mode ~= "Horizontal" then
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.yAxis end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.yAxis end
    end

    Movement.flyV.Velocity = dir.Magnitude > 0 and dir.Unit * Config.Player.FlySpeed or Vector3.zero
    Movement.flyG.CFrame = CFrame.new(r.Position) * cam.CFrame.Rotation
end

function Movement.report()
    local h = Util.humanoid()
    local t = Movement.targetSpeed()
    local d = (h and t) and math.abs(h.WalkSpeed - t) or nil
    return {
        writer = Movement.writerOk,
        detail = Movement.writerOk and "bound" or "unbound",
        holds = t ~= nil and d ~= nil and d <= 0.75,
        delta = t and h and ("target %s actual %s"):format(tostring(t), tostring(h.WalkSpeed)) or "none",
    }
end

function Movement.onSpawn()
    Movement.base = nil
    Movement.current = nil
    task.wait(0.6)
    if Movement.suspended then return end
    if Config.Player.Speed then Movement.applySpeed() end
    if Config.Player.Noclip then Movement.setNoclip(true) end
    if Config.Player.Fly then Movement.startFly() end
    if Config.Player.Bhop then Movement.setBhop(true) end
    if Config.Main.AutoRun then Movement.setAutoRun(true) end
end

function Movement.shutdown()
    Movement.revertSpeed()
    Movement.stopFly()
    Movement.setNoclip(false)
    Movement.setInfJump(false)
    Movement.setBhop(false)
    Movement.setAutoRun(false)
    Movement.unbindWriter()
    Movement.uninstallHook()
    Movement.base = nil
end

Combat = { hookFn = nil, cands = {}, knifeAt = 0, reloadAt = 0 }

local KNIVES = { Knife = true, ["Ancient Knife"] = true, Claymore = true, Crowbar = true }

function Combat.equipped()
    local c = Util.character()
    return c and c:FindFirstChildOfClass("Tool")
end

function Combat.knifeIfEmpty()
    if Combat.equipped() then return end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if not bp then return end
    for _, t in ipairs(bp:GetChildren()) do
        if t:IsA("Tool") and KNIVES[t.Name] then
            local c = Util.character()
            if c then
                for _, verb in ipairs({ "EquipTool", "Equip", "Hotbar" }) do Refs.fire(verb, t.Name) end
                pcall(function() t.Parent = c end)
            end
            return
        end
    end
end

function Combat.knifeNear()
    local t = State.target
    if not t or t.distance > Config.Combat.KnifeRange then return end
    local cur = Combat.equipped()
    if cur and KNIVES[cur.Name] then return end
    Combat.knifeIfEmpty()
end

function Combat.collect()
    table.clear(Combat.cands)
    local cam = Util.cam()
    if not cam then return end
    local origin = cam.CFrame.Position
    local center = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
    local radius = Config.Combat.FovRadius
    local cone = Config.Combat.FovShape == "Cone"

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local c = p.Character
            local h = c and c:FindFirstChildOfClass("Humanoid")
            local head = c and c:FindFirstChild("Head")
            local root = c and c:FindFirstChild("HumanoidRootPart")
            if h and h.Health > 0 and head and root then
                local d = Util.dist(head.Position, origin)
                if d <= Config.Combat.MaxDistance then
                    local scr = Util.screen(head.Position)
                    local cross = scr and (scr - center).Magnitude or radius
                    local inFov = cone
                        and (scr ~= nil and (scr - center).Magnitude <= radius and scr.Y <= center.Y + radius * 0.5)
                        or cross <= radius
                    if inFov and (not Config.Combat.WallCheck or Util.visible(origin, head.Position)) then
                        local w
                        if Config.Combat.TargetMode == "Health" then
                            w = 1 - math.clamp(h.Health / math.max(h.MaxHealth, 1), 0, 1)
                        elseif Config.Combat.TargetMode == "Distance" then
                            w = 1 - math.clamp(d / Config.Combat.MaxDistance, 0, 1)
                        else
                            w = (1 - math.clamp(d / Config.Combat.MaxDistance, 0, 1)) * Config.Combat.WeightDistance
                                + (1 - math.clamp(cross / radius, 0, 1)) * Config.Combat.WeightCrosshair
                        end
                        if w > 0 then
                            Combat.cands[#Combat.cands + 1] = { pos = head.Position, vel = root.AssemblyLinearVelocity,
                                distance = d, weight = w, player = p, health = h.Health }
                        end
                    end
                end
            end
        end
    end
end

function Combat.pick()
    local pool = Combat.cands
    if #pool == 0 then return nil end
    if Config.Combat.TargetMode ~= "Crosshair" then
        local best, bw = nil, -1
        for _, c in ipairs(pool) do
            if c.weight > bw then best, bw = c, c.weight end
        end
        return best
    end
    local total = 0
    for _, c in ipairs(pool) do total += c.weight end
    local roll = math.random() * total
    for _, c in ipairs(pool) do
        roll -= c.weight
        if roll <= 0 then return c end
    end
    return pool[#pool]
end

function Combat.aimFor(c)
    local p = c.pos
    if Config.Combat.Prediction > 0 then
        p += c.vel * ((c.distance / 900) * Config.Combat.Prediction)
    end
    local pad = Config.Combat.HitboxPadding
    if pad > 0 then
        p += Vector3.new((math.random() - 0.5) * pad, (math.random() - 0.5) * pad, (math.random() - 0.5) * pad)
    end
    return p
end

function Combat.centerPoint()
    local cam = Util.cam()
    if not cam then return nil end
    local o = cam.CFrame.Position
    local d = cam.CFrame.LookVector * 1000
    local hit = Util.raycast(o, d)
    return hit and hit.Position or (o + d)
end

function Combat.active()
    if not State.combat or not State.master or Config.Combat.Mode == "Off" then return false end
    if Config.Combat.Trigger == "Hold" then return State.triggerHeld end
    return true
end

function Combat.inject(dt)
    if not Config.Combat.InjectAim or not Config.Combat.Silent then return end
    if not State.target or not State.aimPoint or Combat.hookFn then return end
    State.aimAccum += dt
    if State.aimAccum < 0.033 then return end
    State.aimAccum = 0
    if Refs.UR then pcall(Refs.UR.FireServer, Refs.UR, "ReportAim", State.aimPoint) end
end

function Combat.update(dt)
    if not State.combat or not State.master or Config.Combat.Mode == "Off" then
        State.aimPoint = nil
        State.targetInfo = nil
        return
    end

    local useAim = Combat.active()
    if useAim and (Config.Combat.Mode == "Silent" or Config.Combat.Mode == "Both") then
        Combat.collect()
        State.target = Combat.pick()
    else
        State.target = nil
    end

    local cam = Util.cam()
    if not cam then return end

    if State.target then
        State.aimPoint = Combat.aimFor(State.target)
        State.targetInfo = State.target.player and {
            name = State.target.player.Name,
            health = State.target.health,
            distance = State.target.distance,
        } or nil
    elseif Config.Combat.NoSpread then
        State.aimPoint = Combat.centerPoint()
    else
        State.aimPoint = nil
        State.targetInfo = nil
    end

    if useAim and (Config.Combat.Mode == "Aimbot" or Config.Combat.Mode == "Both") and State.aimPoint then
        cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(cam.CFrame.Position, State.aimPoint),
            math.clamp(Config.Combat.Smoothing, 0.02, 1))
        State.camRot = cam.CFrame.Rotation
    end

    Combat.inject(dt)
end

function Combat.resolve()
    if not Combat.active() or not State.aimPoint then return nil end
    local cam = Util.cam()
    if not cam then return nil end
    State.aimOrigin = cam.CFrame.Position
    return State.aimPoint
end

function Combat.installHook()
    if not HAS_HOOKS or Combat.hookFn then return end
    pcall(function()
        Combat.hookFn = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            if checkcaller() or getnamecallmethod() ~= "FireServer" then return Combat.hookFn(self, ...) end
            if self ~= Refs.RE and self ~= Refs.UR then return Combat.hookFn(self, ...) end
            local args = table.pack(...)
            local act = args[1]
            local pt = Combat.resolve()
            if pt then
                if self == Refs.RE and (act == "M1D" or act == "M2D") then
                    if (Config.Combat.Silent and State.target) or Config.Combat.NoSpread then
                        local o = State.aimOrigin or Util.cam().CFrame.Position
                        args[2] = typeof(args[2]) == "CFrame" and CFrame.lookAt(o, pt) or pt
                        State.lastShot = tick()
                    end
                elseif self == Refs.UR and act == "ReportAim" and Config.Combat.Silent and State.target then
                    args[2] = pt
                elseif self == Refs.RE and act == "Reload" and Config.Combat.AutoReload then
                    Combat.reloadAt = tick()
                end
            end
            return Combat.hookFn(self, table.unpack(args, 1, args.n))
        end))
    end)
end

function Combat.uninstallHook()
    if not Combat.hookFn then return end
    pcall(function() hookmetamethod(game, "__namecall", Combat.hookFn) end)
    Combat.hookFn = nil
end

function Combat.start()
    if State.combat then return end
    State.combat = true
    Combat.installHook()
    Sched.on("knife", true)
    Sched.on("reload", true)
end

function Combat.stop()
    if not State.combat then return end
    State.combat = false
    Combat.uninstallHook()
    Sched.on("knife", false)
    Sched.on("reload", false)
    State.target, State.aimPoint, State.targetInfo = nil, nil, nil
end

Panic = {}

function Panic.isOwned(room)
    for _, a in ipairs({ "Owner", "OwnerId", "OwnerName", "Claimer", "ClaimedBy" }) do
        local v = room:GetAttribute(a)
        if v ~= nil and (v == LocalPlayer.Name or v == LocalPlayer.UserId
            or tostring(v) == tostring(LocalPlayer.UserId)) then return true end
    end
    local cs = room:FindFirstChild("ClaimedStorage")
    if cs then
        for _, d in ipairs(cs:GetDescendants()) do
            if d:IsA("TextLabel") and typeof(d.Text) == "string" and d.Text:find(LocalPlayer.Name, 1, true) then
                return true
            end
        end
    end
    return false
end

function Panic.ownedRoom()
    for _, r in ipairs(Util.rooms()) do
        if Panic.isOwned(r) then return r end
    end
    return nil
end

function Panic.bedIn(room)
    for _, d in ipairs(room:GetDescendants()) do
        if (d:IsA("Model") or d:IsA("BasePart")) and d.Name:lower():find("bed") then return d end
    end
    return nil
end

function Panic.anchor(i)
    if i:IsA("BasePart") then return i.Position end
    local ok, pv = pcall(function() return i:GetPivot().Position end)
    if ok and pv then return pv end
    local p = Util.part(i)
    return p and p.Position or nil
end

function Panic.teleport(dest)
    if not Util.alive() then
        Notify.push("Panic: no character", "bad")
        return
    end
    local room = Panic.ownedRoom()
    if not room then
        Notify.push("Panic: no base", "bad")
        return
    end
    local a
    if dest == "Bed" then
        local bed = Panic.bedIn(room)
        if not bed then
            Notify.push("Panic: no bed", "bad")
            return
        end
        a = Panic.anchor(bed)
    else
        local f = room:FindFirstChild("Floor")
        a = f and f.Position or Panic.anchor(room)
    end
    if not a then
        Notify.push("Panic: no anchor", "bad")
        return
    end
    local c = Util.character()
    if not c then return end
    c:PivotTo(CFrame.new(a + Vector3.new(0, Config.Travel.Height, 0)))
    local r = Util.root()
    if r then r.AssemblyLinearVelocity = Vector3.zero end
    Notify.push(("Panic: %s"):format(dest), "good")
end

World = { backup = nil, fx = nil, fovBack = nil, conn = nil }

local function atmos()
    local a = Lighting:FindFirstChildOfClass("Atmosphere")
    if a then a.Density, a.Haze, a.Glare, a.Offset = 0, 0, 0, 0 end
end

function World.fullbright()
    if World.backup then return end
    World.backup = { Ambient = Lighting.Ambient, OutdoorAmbient = Lighting.OutdoorAmbient,
        Brightness = Lighting.Brightness, ClockTime = Lighting.ClockTime, FogEnd = Lighting.FogEnd }
    World.fx = {}
    for _, e in ipairs(Lighting:GetChildren()) do
        if e:IsA("PostEffect") then
            World.fx[e] = e.Enabled
            e.Enabled = false
        end
    end
    Lighting.Ambient = rgb(170, 170, 170)
    Lighting.OutdoorAmbient = rgb(170, 170, 170)
    Lighting.Brightness = Config.World.Brightness
    Lighting.ClockTime = 14
    Lighting.FogEnd = 1e6
    atmos()
    World.conn = Lighting.ChildAdded:Connect(function(c)
        if c:IsA("Atmosphere") then task.defer(atmos) end
    end)
end

function World.restoreBright()
    if World.conn then
        pcall(function() World.conn:Disconnect() end)
        World.conn = nil
    end
    if World.backup then
        for k, v in pairs(World.backup) do pcall(function() Lighting[k] = v end) end
        World.backup = nil
    end
    if World.fx then
        for e, en in pairs(World.fx) do
            if e.Parent then pcall(function() e.Enabled = en end) end
        end
        World.fx = nil
    end
end

function World.fov()
    local c = Util.cam()
    if not c then return end
    if not World.fovBack then World.fovBack = c.FieldOfView end
    c.FieldOfView = Config.World.FovValue
end

function World.restoreFov()
    local c = Util.cam()
    if c and World.fovBack then c.FieldOfView = World.fovBack end
    World.fovBack = nil
end

function World.resetFov()
    Config.World.FovValue = 70

end

    corner(sw, 4)

    for i, ch in ipairs({ "R", "G", "B" }) do
        local track = Instance.new("Frame")
        track.BackgroundColor3 = Theme.off
        track.Position = UDim2.new(0, 10, 0, 24 + (i - 1) * 12)
        track.Size = UDim2.new(1, -20, 0, 5)
        track.Parent = row
        corner(track, 2)

        fill.BackgroundColor3 = Theme.accent
        fill.Size = UDim2.new(get()[ch], 0, 1, 0)
        fill.Parent = track
        corner(fill, 2)

        local drag = false
        local function apply(x)
            local a = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
            local c = get()
            local n = Color3.new(ch == "R" and a or c.R, ch == "G" and a or c.G, ch == "B" and a or c.B)
            set(n)
            fill.Size = UDim2.new(a, 0, 1, 0)
            sw.BackgroundColor3 = n
        end
        track.InputBegan:Connect(function(inp)
            if inp.UserInputType == Enum.UserInputType.MouseButton1 then
                drag = true
                apply(inp.Position.X)
            end
        end)
        Util.track(UserInputService.InputChanged:Connect(function(inp)
            if drag and inp.UserInputType == Enum.UserInputType.MouseMovement then apply(inp.Position.X) end
        end))
        Util.track(UserInputService.InputEnded:Connect(function(inp)
            if inp.UserInputType == Enum.UserInputType.MouseButton1 then drag = false end
        end))

Menu = { gui = nil, pages = {}, tabs = {} }

local function page(name)
    if not Menu.pages[name] then
        local s = Instance.new("ScrollingFrame")
        s.Size = UDim2.new(1, -16, 1, -16)
        s.Position = UDim2.new(0, 16, 0, 8)
        s.BackgroundTransparency = 1
        s.BorderSizePixel = 0
        s.ScrollBarThickness = 3
        s.CanvasSize = UDim2.new()
        s.AutomaticCanvasSize = Enum.AutomaticSize.Y
        s.Visible = false
        s.Parent = Menu.content
        local l = Instance.new("UIListLayout")
        l.Padding = UDim.new(0, 6)
        l.SortOrder = Enum.SortOrder.LayoutOrder
        l.Parent = s
        Menu.pages[name] = s
    end
    return Menu.pages[name]
end

function Menu.select(name)
    for n, f in pairs(Menu.pages) do f.Visible = n == name end
    for n, b in pairs(Menu.tabs) do
        b.BackgroundColor3 = n == name and Theme.panel or Theme.bg
        b.TextColor3 = n == name and Theme.accent or Theme.muted
    end
end

function Menu.refresh()
    for _, r in ipairs(renders) do pcall(r) end
end

function Menu.search(query)
    local q = query:lower()
    local showAll = q == ""
    for _, row in ipairs(searchables) do
        local txt = row:GetAttribute("SearchText")
        row.Visible = showAll or (type(txt) == "string" and txt:lower():find(q, 1, true) ~= nil)
    end
    for _, p in pairs(Menu.pages) do
        for _, child in ipairs(p:GetChildren()) do
            if child:GetAttribute("IsSection") then child.Visible = showAll end
        end
    end
end

local function flag(section, field, after)
    return function(v)
        Config[section][field] = v
        Cfg.save()
        if after then after(v) end
        Menu.status()
    end
end

function Menu.status()
    if not Menu.st then return end
    local a = {}
    if Config.Player.Speed then a[#a + 1] = "SPD" end
    if Config.Player.Fly then a[#a + 1] = "FLY" end
    if Config.ESP.Mode ~= "Off" then a[#a + 1] = "ESP" end
    if Config.Combat.Mode ~= "Off" then a[#a + 1] = "AIM" end
    if Config.Main.AutoRollSell then a[#a + 1] = "ROLL" end
    Menu.st.Text = not Config.General.Master and "off" or (#a > 0 and table.concat(a, " ") or "ready")
    Menu.st.TextColor3 = Config.General.Master and Theme.accent or Theme.muted
end

function Menu.resync()
    World.restoreBright()
    World.restoreFov()
    if Config.World.Fullbright then World.fullbright() end
    if Config.World.Fov then World.fov() end
    if Config.ESP.Mode == "Native" or Config.ESP.Mode == "Both" then NativeESP.start() else NativeESP.stop() end
    if Config.ESP.Mode == "Overlay" or Config.ESP.Mode == "Both" then Scanner.sweep(); OverlayESP.resync() else OverlayESP.stop() end
    if Config.Combat.Mode == "Off" then Combat.stop() else Combat.start() end
    if Config.Player.Speed then Movement.onSpeed(true) else Movement.onSpeed(false) end
    if Config.Player.Fly then Movement.startFly() else Movement.stopFly() end
    Movement.setNoclip(Config.Player.Noclip)
    Movement.setBhop(Config.Player.Bhop)
    Utility.setRoll(Config.Main.AutoRollSell)
    if Config.Main.InstantOpen or Config.Main.InstantPickup or Config.Main.RemoveDelay then Interact.start() else Interact.stop() end
    Cfg.save()
    Menu.refresh()
    Menu.status()
end

local renderConn
local last = tick()

local function camStep()
    local cam = Util.cam()
    if not cam then return end
    local actual = cam.CFrame.Rotation
    if not State.camRot then
        State.camRot = actual
        return
    end
    if not Config.Combat.NoRecoil or not State.combat or not State.master or Config.Combat.Mode == "Aimbot" then
        State.camRot = actual
        return
    end
    local composed = State.camRot * (actual * State.camRot:Inverse())
    if (tick() - State.lastShot) < 0.35 then composed = State.camRot:Lerp(composed, 0.25) end
    State.camRot = composed
    cam.CFrame = CFrame.new(cam.CFrame.Position) * composed
end

local function start()
    if renderConn then return end
    pcall(function() RunService:UnbindFromRenderStep(CAM_STEP) end)
    pcall(function() RunService:BindToRenderStep(CAM_STEP, Enum.RenderPriority.Camera.Value + 1, camStep) end)
    renderConn = RunService.RenderStepped:Connect(function()
        local now = tick()
        local dt = now - last
        last = now
        Perf.sample(dt)
        if not State.master then return end
        Combat.update(dt)
        State.overlayAccum += dt
        if State.overlayAccum >= 0.033 then
            State.overlayAccum = 0
            OverlayESP.update()
        end
        if Config.Player.Fly then Movement.updateFly() end
    end)
    Sched.start()
end

local function stop()
    if renderConn then
        pcall(function() renderConn:Disconnect() end)
        renderConn = nil
    end
    pcall(function() RunService:UnbindFromRenderStep(CAM_STEP) end)
    Sched.stop()
end

local function inputBegan(input)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        for _, b in ipairs(binds) do
            if b.isListening() then
                b.finish(input.KeyCode)
                Cfg.save()
                return
            end
        end
    end

    if input.KeyCode == Config.General.MenuKey then
        Menu.toggle()
        return
    end

    if input.KeyCode == Config.Combat.TriggerKey then
        State.triggerHeld = true
        return
    end

    if Binds.dispatch(input.KeyCode) then return end

    if input.KeyCode == Config.Player.PanicKey then Panic.teleport(Config.Panic.Destination) end
end

local function inputEnded(input)
    if input.KeyCode == Config.Combat.TriggerKey then State.triggerHeld = false end
    if input.KeyCode == Config.Player.FlyKey then State.flyHeld = false end
end

local function registerBinds()
    Binds.add("speed", function() return Config.Player.SpeedKey end, function()
        Config.Player.Speed = not Config.Player.Speed
        Movement.onSpeed(Config.Player.Speed)
        Notify.push("Speed " .. (Config.Player.Speed and "on" or "off"), Config.Player.Speed and "good" or nil)
        Menu.refresh()
    end)
    Binds.add("noclip", function() return Config.Player.NoclipKey end, function()
        Config.Player.Noclip = not Config.Player.Noclip
        Movement.setNoclip(Config.Player.Noclip)
        Menu.refresh()
    end)
    Binds.add("infjump", function() return Config.Player.InfJumpKey end, function()
        Config.Player.InfJump = not Config.Player.InfJump
        Movement.setInfJump(Config.Player.InfJump)
        Menu.refresh()
    end)
    Binds.add("fly", function() return Config.Player.FlyKey end, function()
        if Config.Player.FlyHold then
            State.flyHeld = true
        else
            Config.Player.Fly = not Config.Player.Fly
            if Config.Player.Fly then Movement.startFly() else Movement.stopFly() end
            Menu.refresh()
        end
    end)
end

local module = {}

local function registerTasks()
    Sched.add("cull", Config.Perf.CullRate, NativeESP.cull)
    Sched.add("tags", 1, NativeESP.applyTags)
    Sched.add("scan", 1.5, function()
        if Config.ESP.Mode == "Overlay" or Config.ESP.Mode == "Both" then
            Scanner.sweep()
            OverlayESP.resync()
        end
    end)
    Sched.add("knife", 0.6, function()
        if not Config.Combat.AutoKnife then return end
        if Config.Combat.KnifeMode == "Near" then Combat.knifeNear() else Combat.knifeIfEmpty() end
    end)
    Sched.add("reload", 0.5, function()
        if not Config.Combat.AutoReload then return end
        local t = Combat.equipped()
        if not t or t:FindFirstChildOfClass("Humanoid") then return end
        local ammo = t:GetAttribute("Ammo") or t:GetAttribute("Clip") or t:GetAttribute("Mag")
        local max = t:GetAttribute("MaxAmmo") or t:GetAttribute("MagSize")
        if type(ammo) == "number" and type(max) == "number" and max > 0 and ammo / max <= Config.Combat.ReloadAt / 10 then
            Refs.fire("Reload")
        end
    end)
    Sched.add("proximity", 3, Alerts.proximity)
    Sched.on("scan", true)
    Sched.on("proximity", true)
end

local function sync()
    State.master = Config.General.Master
    Movement.suspended = false
    registerTasks()
    NativeESP.start()
    if Config.Combat.Mode ~= "Off" then Combat.start() end
    if Config.World.Fullbright then World.fullbright() end
    if Config.World.Fov then World.fov() end
    if Config.Main.InstantOpen or Config.Main.InstantPickup or Config.Main.RemoveDelay then Interact.start() end
    if Config.Main.AntiAfk then Utility.setAfk(true) end
    if Config.Main.AutoRollSell then Utility.setRoll(true) end
    World.spinner(Config.Main.HideSpinner)
    World.chat(Config.Main.ChatWindow)
    Movement.installHook()
    if Config.Player.Speed then Movement.onSpeed(true) end
    if Config.Player.Noclip then Movement.setNoclip(true) end
    if Config.Player.InfJump then Movement.setInfJump(true) end
    if Config.Player.Bhop then Movement.setBhop(true) end
    if Config.Player.Fly then Movement.startFly() end
    if Config.Main.AutoRun then Movement.setAutoRun(true) end
    Alerts.start()
    Menu.status()
end

function module.Unload()
    stop()
    NativeESP.stop()
    OverlayESP.stop()
    Scanner.clear()
    Interact.stop()
    Movement.shutdown()
    Combat.stop()
    Utility.setAfk(false)
    Utility.setRoll(false)
    Alerts.stop()
    World.restoreBright()
    World.restoreFov()
    if Notify.gui then Notify.gui:Destroy() end
    if Menu.gui then Menu.gui:Destroy() end
    Util.untrackAll()
    if ENV.__StorageSuite == module then ENV.__StorageSuite = nil end
    print("[SS] unloaded")
end
