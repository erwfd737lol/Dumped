-- Validate tunables once so malformed settings cannot create infinite loops.
local function clampNumber(key, fallback, minimum, maximum, integer)
    local value = CONFIG[key]
    if type(value) ~= "number" or value ~= value or value == math.huge or value == -math.huge then value = fallback end
    value = math.clamp(value, minimum, maximum)
    if integer then value = math.floor(value) end
    CONFIG[key] = value
end

    local s = game:FindService(name)

    if s then return s end

    return game:GetService(name)

--// ---------- ESP / decoration filter ----------

local IGNORE_PREFIXES = {}

for _, n in ipairs(CONFIG.IGNORE_NAMES) do

    IGNORE_PREFIXES[#IGNORE_PREFIXES + 1] = n:lower()

end



local function isIgnored(inst)

    if not inst then return false end

    local name = inst.Name:lower()

    for i = 1, #IGNORE_PREFIXES do

        local p = IGNORE_PREFIXES[i]

        if name:sub(1, #p) == p then return true end

    end

    return false

end



--// ---------- helpers ----------

-- Accepts plain names AND tables of names (e.g. DOOR_NAMES). Tables are

-- flattened recursively, so callers can pass either form.

local function firstChild(parent, ...)

    if not parent then return nil end

    for i = 1, select("#", ...) do

        local name = select(i, ...)

        if type(name) == "table" then

            local c = firstChild(parent, table.unpack(name))

            if c then return c end

        else

            local c = parent:FindFirstChild(name)

            if c then return c end

        end

    end

    return nil

end



local RootCache = setmetatable({}, { __mode = "k" })

local function getRoot(instance)

    if not instance then return nil end

    local cached = RootCache[instance]

    if cached and cached.Parent then
        if instance:IsA("BasePart") and cached == instance then return cached end
        if cached:IsDescendantOf(instance) then return cached end
    end



    local r

    if instance:IsA("BasePart") then

        r = instance

    elseif instance:IsA("Model") then

        if instance.PrimaryPart then

            r = instance.PrimaryPart

        else

            local fallback

            for _, c in ipairs(instance:GetChildren()) do

                if c:IsA("BasePart") then

                    if isIgnored(c) then

                        fallback = fallback or c

                    else

                        r = c

                        break

                    end

                end

            end

            r = r or fallback or instance:FindFirstChildWhichIsA("BasePart", true)

        end

    else

        r = instance:FindFirstChildWhichIsA("BasePart", true)

    end



    RootCache[instance] = r

    return r

end



local function isAlive()

    return Humanoid ~= nil

       and Humanoid.Health > 0

       and RootPart ~= nil

       and RootPart.Parent ~= nil

end



--// ---------- mob hazard cache ----------

local MobRoots = {}



local function refreshMobCache()
    table.clear(MobRoots)
    local accepted = { Mobs = true, Mob = true, Enemies = true, Monsters = true }
    local seen = {}

    -- Include multiple folders with the same conventional name, whether direct
    -- children of Workspace or nested beneath map containers.
    for _, folder in ipairs(Workspace:GetDescendants()) do
        if accepted[folder.Name] then
            for _, mob in ipairs(folder:GetChildren()) do
                if not seen[mob] then
                    seen[mob] = true
                    local root = getRoot(mob)
                    if root and root.Parent then
                        local hum = mob:FindFirstChildOfClass("Humanoid")
                            or mob:FindFirstChildWhichIsA("Humanoid", true)
                        if not CONFIG.IGNORE_DEAD_MOBS or not hum or hum.Health > 0 then
                            MobRoots[#MobRoots + 1] = root
                        end
                    end
                end
            end
        end
    end
end



-- OPT: squared-distance compare, no sqrt per mob

local function isPositionSafe(position, radius)

    radius = radius or CONFIG.SAFE_RADIUS

    local r2 = radius * radius

    for i = 1, #MobRoots do

        local root = MobRoots[i]

        if root.Parent then

            local d = root.Position - position

            if d.X * d.X + d.Y * d.Y + d.Z * d.Z <= r2 then

                return false

            end

        end

    end

    return true

end

--// ---------- camera ----------

-- capture the player's real FOV the first time we ever see a camera, so the

-- restore can never be skipped because applyGlobals ran before the camera existed

local CameraState = setmetatable({}, { __mode = "k" })

local function captureFOV(cam)
    if not cam or CameraState[cam] then return end
    CameraState[cam] = { fieldOfView = cam.FieldOfView }
end



local function aimAt(target)

    if not CONFIG.AIM_AT or not Camera or not target then return end

    local pos = Camera.CFrame.Position

    local d = pos - target

    if d.X * d.X + d.Y * d.Y + d.Z * d.Z < 1e-6 then return end



    local prev = Camera.CameraType

    Camera.CameraType = Enum.CameraType.Scriptable

    Camera.CFrame = CFrame.lookAt(pos, target)

    if CONFIG.AIM_SETTLE > 0 then task.wait(CONFIG.AIM_SETTLE) end

    if Camera.CameraType == Enum.CameraType.Scriptable then

        Camera.CameraType = prev

    end

end



local function hookCamera(cam)
    if not cam then
        if FovConn then pcall(function() FovConn:Disconnect() end); FovConn = nil end
        Camera = nil
        return
    end

    captureFOV(cam)

    Camera = cam

    if FovConn then

        pcall(function() FovConn:Disconnect() end)

        FovConn = nil

    end

    if RUNNING then cam.FieldOfView = CONFIG.FORCE_FOV end

    FovConn = cam:GetPropertyChangedSignal("FieldOfView"):Connect(function()

        if RUNNING and cam.FieldOfView ~= CONFIG.FORCE_FOV then

            cam.FieldOfView = CONFIG.FORCE_FOV

        end

    end)

end

-- CurrentCamera can be replaced by Roblox during respawn/camera transitions.

    hookCamera(Workspace.CurrentCamera)
end))


--// ---------- prompts ----------

-- originals stored so toggle-off can restore them (OFF means vanilla)

local PatchedPrompts = setmetatable({}, { __mode = "k" })

local fireprompt = fireproximityprompt



local function patchPrompt(prompt)

    if not prompt or PatchedPrompts[prompt] then return end

    PatchedPrompts[prompt] = {

        los  = prompt.RequiresLineOfSight,

        dist = prompt.MaxActivationDistance,

        hold = prompt.HoldDuration,

    }

    prompt.RequiresLineOfSight   = false

    prompt.MaxActivationDistance = CONFIG.PROMPT_RANGE

    prompt.HoldDuration          = 0

end



local function unpatchPrompt(prompt)

    local orig = PatchedPrompts[prompt]

    if not orig then return end

    PatchedPrompts[prompt] = nil

    if prompt.Parent then

        pcall(function()

            prompt.RequiresLineOfSight   = orig.los

            prompt.MaxActivationDistance = orig.dist

            prompt.HoldDuration          = orig.hold

        end)

    end

end



local function unpatchAll()

    for prompt in pairs(PatchedPrompts) do

        unpatchPrompt(prompt)

    end

end



local function Fire(prompt)

    if not prompt or not prompt.Parent then return false end

    patchPrompt(prompt)

    local f = fireprompt or fireproximityprompt

    if not f then return false end

    fireprompt = f
    local ok, err = pcall(f, prompt)
    return ok, err

end



--// ---------- loot pickup wait ----------

local function waitForReparent(item, originalParent, timeout)

    if item.Parent ~= originalParent then return true end



    local done = false

    local conn = item.AncestryChanged:Connect(function()

        if item.Parent ~= originalParent then done = true end

    end)



    local t0 = os.clock()

    while not done and (os.clock() - t0) < timeout do

        task.wait()

    end



    conn:Disconnect()

    return done

end



local function anchorFor(prompt, item)

    local p = prompt.Parent

    if p and p:IsA("BasePart") then return p end

    local r = p and getRoot(p)

    if r and r.Parent then return r end

    return getRoot(item)

end

--// ---------- global mutation (reversible) ----------

local OrigCollide  = setmetatable({}, { __mode = "k" })

local OrigDepth

local DepthApplied = false



local function applyGlobals()

    captureFOV(Camera)

    if Camera then pcall(function() Camera.FieldOfView = CONFIG.FORCE_FOV end) end



    if not DepthApplied then

        OrigDepth = Workspace.FallenPartsDestroyHeight

        DepthApplied = true

    end

    pcall(function() Workspace.FallenPartsDestroyHeight = CONFIG.DROP_DEPTH end)

end



local function restoreGlobals()

    if CONFIG.RESTORE_ON_OFF then

        for cam, original in pairs(CameraState) do
            if cam and cam.Parent and original.fieldOfView then
                pcall(function() cam.FieldOfView = original.fieldOfView end)
            end
            CameraState[cam] = nil
        end

        if DepthApplied and OrigDepth ~= nil then
            pcall(function() Workspace.FallenPartsDestroyHeight = OrigDepth end)
            DepthApplied = false
            OrigDepth = nil
        end

        for part, orig in pairs(OrigCollide) do

            if part.Parent then pcall(function() part.CanCollide = orig end) end

        end

        unpatchAll()

    end



    -- always clear, even when RESTORE_ON_OFF is false, so the map can't grow

    table.clear(OrigCollide)

end



local function setRunning(on)

    if RUNNING == on then return end

    RUNNING = on

    if on then

        applyGlobals()

        applyHealth()

    else

        restoreGlobals()

        restoreHealth()

    end

    if uiRefresh then uiRefresh() end

end



--// ---------- character bind ----------

local function bindCharacter(newChar)
    if not newChar then return end
    local previousHumanoid = Humanoid
    if previousHumanoid then restoreHealth(previousHumanoid) end

    Character = newChar
    Humanoid = newChar:WaitForChild("Humanoid", 5)
    RootPart = newChar:WaitForChild("HumanoidRootPart", 5)



    table.clear(OrigCollide)
    rebuildCharParts()

    hookCamera(Workspace.CurrentCamera)



    for _, c in ipairs(CharConns) do pcall(function() c:Disconnect() end) end

    table.clear(CharConns)



    local function addConn(c)

        CharConns[#CharConns + 1] = c

    end



    addConn(newChar.DescendantAdded:Connect(function()
        rebuildCharParts()
    end))
    addConn(newChar.DescendantRemoving:Connect(function()
        rebuildCharParts()
    end))



    if Humanoid then

        -- No client-side MaxHealth enforcement; server-authoritative health is left alone.

        addConn(Humanoid.Died:Connect(function()

            if RUNNING then
                Stats.deaths += 1
                if uiRefresh then pcall(uiRefresh) end
            end

        end))

    end



    if RUNNING then applyHealth() end

end



--// ---------- root discovery ----------

-- direct child first, then anywhere in Workspace, so a BasementStorages nested

-- inside Storages is still found

local function findFolder(name)
    local direct = Workspace:FindFirstChild(name)
    if direct then return direct end
    -- Prefer a candidate that actually contains room-like children over an
    -- unrelated duplicate with the same name.
    local fallback
    for _, candidate in ipairs(Workspace:GetDescendants()) do
        if candidate.Name == name then
            fallback = fallback or candidate
            if candidate:FindFirstChild("Contents")
                or candidate:FindFirstChild("Room")
                or candidate:FindFirstChild("BasementRoom")
                or candidate:FindFirstChild("StorageRoom") then
                return candidate
            end
        end
    end
    return fallback
end



-- drop any root nested inside another root, so nothing is walked twice

local function pruneNested(list)

    local out = {}

    for i = 1, #list do

        local nested = false

        for j = 1, #list do

            if i ~= j and list[i]:IsDescendantOf(list[j]) then

                nested = true

                break

            end

        end

        if not nested then out[#out + 1] = list[i] end

    end

    return out

end



-- cached: a recursive FindFirstChild every 0.5s is wasteful

local Roots, RootsAt = {}, 0



local function resolveRoots(now)

    local valid = #Roots > 0

    if valid then

        for i = 1, #Roots do

            if not Roots[i].Parent then valid = false; break end

        end

    end

    if valid and (now - RootsAt) < CONFIG.ROOT_RESOLVE_INTERVAL then

        return Roots

    end



    local list = {}

    for _, n in ipairs(CONFIG.ROOT_NAMES) do

        local f = findFolder(n)

        if f then list[#list + 1] = f end

    end



    if #list == 0 then

        for _, n in ipairs(CONFIG.ROOT_FALLBACK) do

            local f = findFolder(n)

            if f then list[#list + 1] = f end

        end

        if #list > 0 then

            warnOnce("fallback", "[Farm] BasementStorages not found — using fallback roots")

        end

    end



    Roots   = pruneNested(list)

    RootsAt = now

    return Roots

end



--// ---------- structure discovery ----------

-- a CLOSED room has no Contents folder, so requiring one would skip exactly the

-- rooms that need a door. Recurse through wrappers so a room nested two levels

-- down is still found, instead of the wrapper being pushed as a fake room.

local function expandStorages(node, out, depth)
    if not node then return end
    depth = depth or 0

    if node:FindFirstChild("Contents") then
        out[#out + 1] = node
        return
    end

    -- Closed rooms may lack Contents, but must have a recognized door or room
    -- marker. Do not promote arbitrary empty wrappers into fake rooms.
    local hasDoor = firstChild(node, "Door", "BasementDoor", "DoorPart") ~= nil
    local roomMarked = node.Name == "Room" or node.Name == "BasementRoom"
        or node.Name == "StorageRoom"
    if hasDoor or (roomMarked and #node:GetChildren() == 0) then
        out[#out + 1] = node
        return
    end

    if depth >= CONFIG.MAX_EXPAND_DEPTH then return end
    for _, child in ipairs(node:GetChildren()) do
        expandStorages(child, out, depth + 1)
    end
end



-- find the room by DOOR PRESENCE, not by name — handles "Room", "BasementRoom"

-- and anything else

local DOOR_NAMES = { "Door", "BasementDoor", "DoorPart" }

local ROOM_NAMES = { "Room", "BasementRoom", "StorageRoom" }



local function findDoorFor(storage)

    local node = storage

    for _ = 1, 4 do

        if not node or node == Workspace then break end



        local direct = firstChild(node, DOOR_NAMES)

        if direct then return direct end



        for _, n in ipairs(ROOM_NAMES) do

            local named = node:FindFirstChild(n)

            if named then

                local nd = firstChild(named, DOOR_NAMES)

                if nd then return nd end

            end

        end



        node = node.Parent

    end

    return nil

end



local function countReal(folder)

    if not folder then return 0 end

    local n = 0

    for _, c in ipairs(folder:GetChildren()) do

        if not isIgnored(c) then n += 1 end

    end

    return n

end



--// ---------- door reaction test ----------

local function doorReacted(doorRoot, prompt, beforeCF, beforeAlpha, beforeEnabled, beforeCanCollide)
    if not doorRoot or not doorRoot.Parent then return true end
    if doorRoot.CFrame ~= beforeCF then return true end
    if doorRoot.Transparency ~= beforeAlpha then return true end
    if doorRoot.CanCollide ~= beforeCanCollide then return true end
    if prompt and beforeEnabled == true and prompt.Enabled == false then return true end
    return false
end

local function pollDoor(doorRoot, prompt, beforeCF, beforeAlpha, beforeEnabled, beforeCanCollide, timeout, deadline)
    local t0 = os.clock()
    while (os.clock() - t0) < timeout do
        if not RUNNING or not isAlive() then return false end
        if deadline and os.clock() >= deadline then return false end
        if doorReacted(doorRoot, prompt, beforeCF, beforeAlpha, beforeEnabled, beforeCanCollide) then return true end
        task.wait(0.03)
    end
    return doorReacted(doorRoot, prompt, beforeCF, beforeAlpha, beforeEnabled, beforeCanCollide)
end


--// ---------- room work ----------

-- consecutive unsafe passes per item, so a room isn't stalled forever by a

-- parked mob just inside SAFE_RADIUS

local UnsafeTries = setmetatable({}, { __mode = "k" })



local function lootStorage(storage, deadline)

    local contents = firstChild(storage, "Contents")

    local loot = contents and firstChild(contents, "Loot")

    if not loot then return 0 end



    local taken = 0

    for _, item in ipairs(loot:GetChildren()) do

        if not RUNNING or not isAlive() then break end

        if os.clock() > deadline then break end



        if not isIgnored(item) then

            local prompt = item:FindFirstChildWhichIsA("ProximityPrompt", true)



            if prompt then

                local root = RootPart

                if not root or not root.Parent then break end



                local anchor = anchorFor(prompt, item)

                if anchor and anchor.Parent then

                    local safe = isPositionSafe(anchor.Position)

                    if not safe then

                        local n = (UnsafeTries[item] or 0) + 1

                        UnsafeTries[item] = n

                        safe = n >= CONFIG.FORCE_LOOT_AFTER

                    else

                        UnsafeTries[item] = 0

                    end



                    if safe then

                        local originalParent = item.Parent



                        root.CFrame = anchor.CFrame + Vector3.new(0, 3, 0)

                        root.AssemblyLinearVelocity = ZERO

                        task.wait(0.01)

                        aimAt(anchor.Position)



                        -- SPAM: keep re-firing until the item actually leaves the

                        -- loot folder, we die, or we run out of time

                        local lootDeadline = math.min(deadline, os.clock() + CONFIG.LOOT_TIMEOUT)
                        while os.clock() < lootDeadline do
                            if item.Parent ~= originalParent then break end
                            if not RUNNING or not isAlive() or os.clock() >= deadline then break end
                            Fire(prompt)
                            if waitForReparent(item, originalParent, math.min(CONFIG.PROMPT_SPAM_DELAY, math.max(0, lootDeadline - os.clock()))) then
                                break
                            end
                        end

                        if item.Parent ~= originalParent then taken += 1 end

                    end

                end

            end

        end

    end



    return taken

end



local function openDoor(door, deadline)
    if not door or not RUNNING or not isAlive() or os.clock() >= deadline then return false end



    local doorRoot = getRoot(door)

    if not doorRoot or not doorRoot.Parent then return false end

    if not isPositionSafe(doorRoot.Position) then return false end



    local prompt = door:FindFirstChildWhichIsA("ProximityPrompt", true)

    if not prompt then

        if CONFIG.DEBUG then warnOnce(door, "[Farm] door has no prompt:", door.Name) end

        return false

    end



    local base = doorRoot.CFrame + (doorRoot.CFrame.LookVector * 0.5)



    if CONFIG.DOOR_PHASE_DROP then

        RootPart.CFrame = base + Vector3.new(0, CONFIG.DOOR_PHASE_OFFSET, 0)

        RootPart.AssemblyLinearVelocity = ZERO

        task.wait(math.min(CONFIG.DOOR_PRELOAD_WAIT, math.max(0, deadline - os.clock())))

        if not isAlive() or not RUNNING or os.clock() >= deadline then return false end

    end



    RootPart.CFrame = base + Vector3.new(0, 3, 0)

    RootPart.AssemblyLinearVelocity = ZERO

    task.wait(math.min(CONFIG.DOOR_ARRIVE_WAIT, math.max(0, deadline - os.clock())))

    if not isAlive() or not RUNNING or os.clock() >= deadline then return false end



    aimAt(doorRoot.Position)



    local beforeCF, beforeAlpha = doorRoot.CFrame, doorRoot.Transparency
    local beforeEnabled, beforeCanCollide = prompt.Enabled, doorRoot.CanCollide
    Fire(prompt)
    local opened = pollDoor(doorRoot, prompt, beforeCF, beforeAlpha, beforeEnabled, beforeCanCollide, CONFIG.DOOR_CONFIRM, deadline)

    if not opened and RUNNING and isAlive() and os.clock() < deadline then
        Fire(prompt)
        opened = pollDoor(doorRoot, prompt, beforeCF, beforeAlpha, beforeEnabled, beforeCanCollide, CONFIG.DOOR_SETTLE, deadline)
    end

    if not opened then
        if CONFIG.DEBUG then warnOnce(door, "[Farm] door activation was not confirmed:", door.Name) end
        return false
    end
    if not isAlive() or not RUNNING or os.clock() >= deadline then return false end
    task.wait(math.min(CONFIG.DOOR_AFTER_WAIT, math.max(0, deadline - os.clock())))

    if CONFIG.DOOR_EXIT_DROP and isAlive() and RUNNING and os.clock() < deadline then
        RootPart.CFrame = base + Vector3.new(0, CONFIG.DOOR_EXIT_OFFSET, 0)
        RootPart.AssemblyLinearVelocity = ZERO
        task.wait(math.min(CONFIG.DOOR_DROP_WAIT, math.max(0, deadline - os.clock())))
    end
    return true

end



--// ---------- room skip-cache ----------

local StorageState = setmetatable({}, { __mode = "k" })



-- returns (lootCount, objectCount); a pair, not a packed number, so

-- "512 loot / 0 objects" no longer collides with "0 loot / 1 object"

local function folderSignature(folder)
    if not folder then return "<missing>" end
    local parts = {}
    for _, child in ipairs(folder:GetChildren()) do
        if not isIgnored(child) then
            local identity = child.Name .. ":" .. child.ClassName
            -- GetDebugId is not available in every environment; fall back to
            -- structural properties while still including prompt state.
            local ok, id = pcall(function() return child:GetDebugId(0) end)
            if ok and id then identity ..= ":" .. tostring(id) end
            local prompt = child:FindFirstChildWhichIsA("ProximityPrompt", true)
            if prompt then identity ..= ":P" .. tostring(prompt.Enabled) .. ":" .. tostring(prompt.HoldDuration) end
            parts[#parts + 1] = identity
        end
    end
    table.sort(parts)
    return table.concat(parts, "|")
end

local function storageSig(storage)
    local contents = firstChild(storage, "Contents")
    if not contents then return "<no-contents>", "<no-contents>" end
    local loot = firstChild(contents, "Loot")
    local objects = firstChild(contents, "Objects")
    return folderSignature(loot), folderSignature(objects)
end

-- Reparenting, prompt changes, and streaming can alter room state without
-- changing the child counts. Invalidate the room cache on hierarchy events.
track(Workspace.DescendantAdded:Connect(function() table.clear(StorageState) end))
track(Workspace.DescendantRemoving:Connect(function() table.clear(StorageState) end))


--// ---------- main loop ----------

task.spawn(function()
    local warnedNoStorages = false
    local lastMobRefresh = -math.huge



    while true do

        task.wait(CONFIG.LOOP_DELAY)

        if DEAD then break end



        if uiRefresh then uiRefresh() end

        if not RUNNING or not isAlive() then continue end



        local now = os.clock()
        if now - lastMobRefresh >= 1.0 then
            local okMob = pcall(refreshMobCache)
            if not okMob then warnOnce("mobcache", "[Farm] refreshMobCache failed") end
            lastMobRefresh = now
        end

        local roots = resolveRoots(now)

        if #roots == 0 then

            if not warnedNoStorages then

                warnedNoStorages = true

                warn("[Farm] no storages folder found — streaming may still be enabled")

            end

            continue

        end

        warnedNoStorages = false



        if CONFIG.DEBUG then

            local names = {}

            for i, r in ipairs(roots) do names[i] = r:GetFullName() end

            local key = "roots:" .. table.concat(names, ",")

            warnOnce(key, "[Farm] farming:", table.concat(names, ", "))

        end



        local lootPass, doorPass = {}, {}

        local seenDoor = {}



        local okScan, scanErr = pcall(function()

            local nodes = {}

            for _, rootFolder in ipairs(roots) do

                for _, child in ipairs(rootFolder:GetChildren()) do

                    expandStorages(child, nodes)

                end

            end



            for _, storage in ipairs(nodes) do

                local st = StorageState[storage]

                local sl, so = storageSig(storage)



                if st and st.l == sl and st.o == so and (now - st.last) < CONFIG.RECHECK_AFTER then

                    Stats.skipped += 1

                    continue

                end



                local contents = firstChild(storage, "Contents")

                local loot     = contents and firstChild(contents, "Loot")

                local objects  = contents and firstChild(contents, "Objects")



                local hasLoot = countReal(loot) > 0

                local isEmpty = countReal(objects) == 0



                if hasLoot then

                    lootPass[#lootPass + 1] = { key = storage, node = storage }

                end



                if isEmpty then

                    local door = findDoorFor(storage)

                    if door then

                        if not seenDoor[door] then

                            seenDoor[door] = true

                            doorPass[#doorPass + 1] = { key = storage, door = door }

                        end

                    elseif CONFIG.DEBUG then

                        warnOnce(storage, "[Farm] empty room with no door:", storage.Name)

                    end

                end



                if not hasLoot and not isEmpty then

                    StorageState[storage] = { l = sl, o = so, last = now }

                end

            end

        end)

        if not okScan then

            warnOnce("scan", "[Farm] scan failed:", scanErr)

        end



        for _, entry in ipairs(lootPass) do

            if not RUNNING or not isAlive() then break end



            local ok, taken = pcall(lootStorage, entry.node, os.clock() + CONFIG.ROOM_BUDGET)

            if ok then

                Stats.items += taken or 0
                if uiRefresh then pcall(uiRefresh) end

            else
                Stats.lootFailures += 1
                if uiRefresh then pcall(uiRefresh) end
                warnOnce(entry.key, "[Farm] loot failed:", entry.key.Name, taken)
            end

            local l, o = storageSig(entry.key)

            StorageState[entry.key] = { l = l, o = o, last = os.clock() }

        end



        for _, entry in ipairs(doorPass) do

            if not RUNNING or not isAlive() then break end



            Stats.doorAttempts += 1
            local ok, opened = pcall(openDoor, entry.door, os.clock() + CONFIG.ROOM_BUDGET)

            if ok and opened then

                Index += 1

                Stats.rooms += 1
                if uiRefresh then pcall(uiRefresh) end

                if CONFIG.RESET_ENABLED

                   and CONFIG.RESET_EVERY > 0
                   and Index % CONFIG.RESET_EVERY == 0

                   and isAlive() then

                    RootPart.CFrame = CFrame.new(CONFIG.RESET_POS)

                    RootPart.AssemblyLinearVelocity = ZERO

                    task.wait(0.75)

                    table.clear(StorageState)

                end

            else
                Stats.doorFailures += 1
                if uiRefresh then pcall(uiRefresh) end
                if not ok then
                    warnOnce(entry.door, "[Farm] door failed:", entry.door.Name, opened)
                end
            end



            local l, o = storageSig(entry.key)

            StorageState[entry.key] = { l = l, o = o, last = os.clock() }

        end

    end

end)

local function destroy()
    if DEAD then return end
    RUNNING = false
    DEAD = true



    for _, c in ipairs(Connections) do pcall(function() c:Disconnect() end) end

    table.clear(Connections)

    for _, c in ipairs(CharConns) do pcall(function() c:Disconnect() end) end

    table.clear(CharConns)

    if FovConn then pcall(function() FovConn:Disconnect() end); FovConn = nil end

    pcall(restoreGlobals)
    pcall(restoreHealth)
    table.clear(HealthState)

    pcall(unpatchAll)



    table.clear(StorageState)

    table.clear(MobRoots)

    table.clear(UnsafeTries)



    if GUI then pcall(function() GUI:Destroy() end); GUI = nil end
    Character, RootPart, Humanoid, Camera = nil, nil, nil, nil

    if ENV.__BasementFarm and ENV.__BasementFarm.destroy == destroy then

        ENV.__BasementFarm = nil

    end

end
